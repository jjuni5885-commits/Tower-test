# 폰 손가락 탭으로만 처음부터 끝까지 눌러 보는 시험: python3 tools/touchflow.py [가로x세로]
# 버튼은 화면 좌표를 구해 touchscreen.tap 으로 누르고, 누른 자리 맨 위 요소가 그 버튼인지도 확인
import sys, asyncio
from playwright.async_api import async_playwright
import os
ROOT = os.path.abspath(os.path.join(os.path.dirname(os.path.abspath(__file__)), '..'))
OUT = ROOT + '/shots/'
RES = []
def ok(name, cond, info=''):
    RES.append((name, bool(cond)))
    print(('통과 ' if cond else '실패 ') + name + ('  ' + str(info) if info != '' else ''))
async def main():
    W, H = 360, 640
    if len(sys.argv) > 1 and 'x' in sys.argv[1]: W, H = map(int, sys.argv[1].split('x'))
    async with async_playwright() as p:
        b = await p.chromium.launch()
        ctx = await b.new_context(viewport={'width': W, 'height': H}, device_scale_factor=3, is_mobile=True, has_touch=True)
        pg = await ctx.new_page()
        errs = []
        pg.on('pageerror', lambda e: errs.append('PAGE: ' + str(e)))
        pg.on('console', lambda m: errs.append(m.type + ': ' + m.text) if m.type == 'error' else None)
        await pg.goto('file://' + ROOT + '/out/index.html')
        await pg.wait_for_timeout(1500)
        async def ev(js): return await pg.evaluate(js)
        async def tap_xy(x, y, wait=350):
            await pg.touchscreen.tap(x, y); await pg.wait_for_timeout(wait)
        async def tap_el(sel, wait=400, idx=0):
            r = await ev("""(function(){var els=document.querySelectorAll(%r); var e=els[%d]; if(!e) return null;
              var b=e.getBoundingClientRect(); if (b.top < 0 || b.bottom > innerHeight) { e.scrollIntoView({block:'center'}); b=e.getBoundingClientRect(); }
              var x=b.left+b.width/2, y=b.top+b.height/2, t=document.elementFromPoint(x,y);
              return {x:x, y:y, hit: !!t && (e===t || e.contains(t)), top: t ? (t.id || t.className || t.tagName) : null};})()""" % (sel, idx))
            if not r: return {'missing': sel}
            await tap_xy(r['x'], r['y'], wait)
            return r
        async def cell_xy(c, r):
            return await ev("(function(){var v=__td.view(); return [(v.ox + (%d + 0.5) * v.S) / v.dpr, (v.oy + (%d + 0.5) * v.S) / v.dpr];})()" % (c, r))
        async def free_cells():
            return await ev("""(function(){var g=__td.G(), out=[], GC=g.cols, GR=g.rows; for (var r=1;r<GR-1;r++) for (var c=0;c<GC;c++){ var i=r*GC+c;
               if (g.kind[i]!=='g' || g.obsAt[i]>=0) continue; if (g.towers.some(function(t){return t.c===c&&t.r===r;})) continue;
               var near=false; for (var dr=-1;dr<=1;dr++) for (var dc=-1;dc<=1;dc++){ var rr=r+dr, cc=c+dc; if (rr>=0&&rr<GR&&cc>=0&&cc<GC&&g.kind[rr*GC+cc]==='p') near=true; }
               if (near) out.push([c,r]); } return out;})()""")
        async def cards():
            n = 0
            while await ev("!!document.getElementById('bOk') && !document.getElementById('sModal').classList.contains('hide')") and n < 6:
                r = await tap_el('#bOk', 400); n += 1
                ok('새 적 카드 확인 탭', not await ev("!!document.getElementById('bOk') && !document.getElementById('sModal').classList.contains('hide')") or n > 1, r)
            return n
        def vis(id): return "(function(){var e=document.getElementById('%s'); return !!e && !e.classList.contains('hide') && getComputedStyle(e).display!=='none';})()" % id

        # 1. 타이틀: 버전 글자(관리자 입구)를 눌러도 넘어가지 않음 → 화면 탭 → 챕터 고르기 (v2.0)
        r = await tap_el('#ver', 400)
        ok('버전 글자 탭은 타이틀에 머묾', await ev(vis('sTitle')), r)
        await tap_xy(W / 2, H / 2, 1200)
        ok('타이틀 탭 → 챕터 고르기', await ev(vis('sMap')) and await ev(vis('chSel')) and await ev("document.body.getAttribute('data-ch')") == '0')
        await pg.screenshot(path=OUT + 'tf_chsel.png')
        # 1-1. 잠긴 챕터 카드는 안내만, 1장 카드 → 챕터 지도
        nlock = await ev("document.querySelectorAll('.chCard.locked').length")
        if nlock:
            r = await tap_el('.chCard.locked', 500)
            ok('잠긴 챕터 카드는 안내만', await ev(vis('chSel')) and await ev("document.getElementById('toast').classList.contains('on')"), r)
        r = await tap_el('.chCard', 900)
        n = 0
        while await ev(vis('story')) and n < 20:   # 챕터 시작 이야기가 있으면 넘김
            await tap_xy(W / 2, H * 0.78, 300); n += 1
        ok('1장 카드 탭 → 챕터 지도', await ev("__ch.st().mode") == 'map' and not await ev(vis('chSel')) and await ev("document.body.getAttribute('data-ch')") == '4' and await ev("document.querySelectorAll('.node').length") > 0, r)
        await pg.screenshot(path=OUT + 'tf_chmap.png')
        # 1-2. 챕터 단추 → 챕터 고르기 → 다시 1장
        r = await tap_el('#mBack', 700)
        ok('챕터 단추 → 챕터 고르기', await ev(vis('chSel')), r)
        r = await tap_el('.chCard', 900)
        ok('다시 1장 지도', await ev("__ch.st().mode") == 'map', r)
        # 2. 1번 노드 → 판 창
        r = await tap_el('.node', 800)
        ok('1번 노드 탭 → 지역 창', await ev(vis('sModal')), r)
        await pg.screenshot(path=OUT + 'tf_stage.png')
        # 2-1. 출전 편성: 편성 창 → 탑 하나 빼고 다시 넣기 → 확인 → 판 창으로
        r = await tap_el('#stLo', 700)
        ok('편성 칸 탭 → 편성 창', await ev("__td.st().kind") == 'lo', r)
        n0 = await ev("document.querySelectorAll('.loT.in').length")
        r = await tap_el('.loT.in', 400)
        n1 = await ev("document.querySelectorAll('.loT.in').length")
        r2 = await tap_el('.loT:not(.in):not(.off)', 400)
        n2 = await ev("document.querySelectorAll('.loT.in').length")
        ok('편성에서 탑 빼기, 넣기', n1 == n0 - 1 and n2 == n0, [n0, n1, n2, r, r2])
        await pg.screenshot(path=OUT + 'tf_lo.png')
        r = await tap_el('#loDet', 600)
        ok('상세 단추 → 탑 상세', await ev("__td.st().kind") == 'twd', r)
        r = await tap_el('#tdBack', 600)
        ok('탑 상세 → 편성 창', await ev("__td.st().kind") == 'lo', r)
        r = await tap_el('#loOk', 700)
        ok('편성 확인 → 판 창', await ev("__td.st().kind") == 'stage' and await ev("__lo.get(0).length") == n0, r)
        # 3. 난이도 탭 (쉬움 → 보통)
        r = await tap_el('.tab[data-dk="easy"]', 500)
        ok('쉬움 탭', await ev("!!document.querySelector('.tab.on[data-dk=\"easy\"]')"), r)
        r = await tap_el('.tab[data-dk="normal"]', 500)
        ok('보통 탭', await ev("!!document.querySelector('.tab.on[data-dk=\"normal\"]')"), r)
        # 4. 출전 → 이야기
        r = await tap_el('#bGo', 900)
        ok('출전 탭 → 이야기', await ev(vis('story')), r)
        await pg.screenshot(path=OUT + 'tf_story.png')
        n = 0
        while await ev(vis('story')) and n < 20:
            await tap_xy(W / 2, H * 0.78, 300); n += 1
        ok('이야기 탭으로 넘기기', not await ev(vis('story')), '탭 %d번' % n)
        await pg.wait_for_timeout(700)
        ok('전투 시작', await ev("!!__td.G() && __td.G().mi===0"))
        ok('전투에 편성한 탑이 넘어감', await ev("JSON.stringify(__td.G().towersOk || [])") == await ev("JSON.stringify(__tf.loadout(0))") and await ev("document.body.getAttribute('data-ch')") == '4', await ev("JSON.stringify(__td.G().towersOk || [])"))
        await pg.screenshot(path=OUT + 'tf_tut1.png')
        print('   안내:', await ev("document.getElementById('tut').textContent"))
        # 5. 탑 짓기: 빈 칸 탭 → 궁수탑 카드 두 번
        cells = await free_cells()
        c, rr = cells[len(cells) // 2]
        x, y = await cell_xy(c, rr)
        await tap_xy(x, y, 500)
        ok('빈 칸 탭 → 건설 쟁반', await ev("document.getElementById('tray').className.indexOf('build')>=0"))
        await pg.screenshot(path=OUT + 'tf_tray.png')
        r1 = await tap_el('.tc[data-k="sling"]', 400)
        r2 = await tap_el('.tc[data-k="sling"]', 500)
        ok('궁수탑 카드 두 번 → 건설', await ev("__td.G().towers.length") == 1, [r1, r2])
        # 두 번째 탑(투석기)
        cells = await free_cells()
        c2, r2c = cells[0]
        x2, y2 = await cell_xy(c2, r2c)
        await tap_xy(x2, y2, 500)
        await tap_el('.tc[data-k="spear"]', 400); await tap_el('.tgo', 500)
        ok('투석기 건설 (카드 한 번 + 건설 단추)', await ev("__td.G().towers.length") == 2 and await ev("__td.st().modal") == False)
        print('   안내:', await ev("document.getElementById('tut').textContent"))
        # 6. 탑 강화: 탑 탭 → 강화 카드 두 번
        await tap_xy(x, y, 500)
        ok('탑 탭 → 강화 쟁반', await ev("document.getElementById('tray').className.indexOf('tower')>=0"))
        await pg.screenshot(path=OUT + 'tf_up.png')
        await tap_el('.uc', 400); await tap_el('.uc', 500)
        ok('강화 카드 두 번 → 2단계', await ev("__td.G().towers.filter(function(t){return t.k==='sling';})[0].lvl") == 2)
        print('   안내:', await ev("document.getElementById('tut').textContent"))
        # 쟁반 닫기 (빈 땅 탭)
        if await ev("document.getElementById('tray').className.indexOf('hide')<0"):
            await tap_el('#tray [data-a="close"]', 400)
        # 7. 웨이브 시작 버튼
        r = await tap_el('#bWave', 600)
        ok('전투 시작 버튼 → 웨이브 1', await ev("__td.G().wave") == 1, r)
        await pg.wait_for_timeout(1500)
        await pg.screenshot(path=OUT + 'tf_card.png')
        await cards()
        # 8. 속도
        r = await tap_el('#bSpeed', 300)
        ok('속도 버튼 → 2배', '2' in await ev("document.getElementById('spdTxt').textContent"), r)
        # 9. 일시정지 → 계속하기
        await cards()
        r = await tap_el('#bPause', 500)
        ok('일시정지 탭 → 창', await ev(vis('sModal')), r)
        r = await tap_el('#pRes', 500)
        ok('계속하기', not await ev(vis('sModal')), r)
        # 10. 화살비: 버튼 → 길 위 탭
        await ev("__td.G().wave")
        await cards()
        r = await tap_el('#sk_rain', 400)
        pc = await ev("(function(){var g=__td.G(); var p=g.paths[0]; var k=Math.floor(p.n*0.4); return [p.xs[k], p.ys[k]];})()")
        vx = await ev("(function(){var v=__td.view(); return [(v.ox + %f * v.S) / v.dpr, (v.oy + %f * v.S) / v.dpr];})()" % (pc[0], pc[1]))
        await tap_xy(vx[0], vx[1], 500)
        ok('화살비 사용', await ev("__td.G().sk.rain.cd") > 0, r)
        # 11. 끝까지 빨리 돌리고 결과 창
        await ev("__td.G().gold=3000; __td.bot(1); for (var i=0;i<600 && !__td.G().over;i++) __td.step(60);")
        await pg.wait_for_timeout(500)
        await cards()
        # 결과 창은 승리 연출(1.6초) 뒤에 뜬다. 시험 기계가 느릴 때를 대비해 20초까지 기다림
        for i in range(80):
            await cards()
            if await ev("__td.st().kind==='result'"): break
            await pg.wait_for_timeout(250)
        ok('결과 창', await ev(vis('sModal')) and await ev("__td.st().kind") == 'result', await ev("[__td.G().over, __td.G().lives]"))
        await pg.screenshot(path=OUT + 'tf_result.png')
        for i in range(6):
            if await ev("!!document.getElementById('rMap')"): break
            await pg.wait_for_timeout(500)
        await pg.wait_for_timeout(1200)   # 새 탑 획득 창(결과 창 0.9초 뒤)이 뜨면 사람처럼 먼저 확인을 누름
        if await ev("!!document.getElementById('rwOk')"): await tap_el('#rwOk', 600)
        r = await tap_el('#rMap', 1200)
        if not await ev(vis('story')):
            ok('결과 → 지도', await ev(vis('sMap')) and not await ev(vis('sModal')), r)
        else:
            n = 0
            while await ev(vis('story')) and n < 20: await tap_xy(W / 2, H * 0.78, 300); n += 1
            ok('결과 → (이야기) → 지도', await ev(vis('sMap')), r)
        # 12. 연구
        r = await tap_el('#bRes', 600)
        ok('연구 창', await ev(vis('sModal')), r)
        await pg.screenshot(path=OUT + 'tf_res.png')
        before = await ev("JSON.stringify(__td.SAVE().res)")
        r = await tap_el('.ritem.can', 400)
        r2 = await tap_el('.ritem.can', 500) if await ev("!!document.querySelector('.ritem.can')") else None
        after = await ev("JSON.stringify(__td.SAVE().res)")
        print('   연구 전후:', before, after, r)
        cl = '#rDone' if await ev("!!document.getElementById('rDone')") else '.xbtn'
        r = await tap_el(cl, 500)
        ok('연구 닫기', not await ev(vis('sModal')), r)
        # 13. 도감
        r = await tap_el('#bCodex', 600)
        ok('도감 창', await ev(vis('sModal')), r)
        r = await tap_el('.ci', 400, 1)
        await pg.screenshot(path=OUT + 'tf_codex.png')
        r = await tap_el('#cClose', 500)
        ok('도감 닫기', not await ev(vis('sModal')), r)
        # 14. 설정
        r = await tap_el('#bSet', 600)
        ok('설정 창', await ev(vis('sModal')), r)
        r = await tap_el('#sClose', 500)
        ok('설정 닫기', not await ev(vis('sModal')), r)
        # 15. 2번 노드 열기, 닫기
        r = await tap_el('.node', 700, 1)
        ok('2번 노드 → 지역 창', await ev(vis('sModal')), r)
        r = await tap_el('#bClose', 500)
        ok('지역 창 닫기', not await ev(vis('sModal')), r)
        print('오류:', errs[:10])
        bad = [n for n, c in RES if not c]
        print('결과: %d/%d 통과' % (len(RES) - len(bad), len(RES)), '실패:', bad)
        await b.close()
asyncio.run(main())
