# 소환, 확률 보기, 소환 결과(1회, 10회), 시작 능력 카드, 능력 카드(조합 배지), 빌드 보기, 조합 연출, 지도 2부(맵 20개 흉내), 지역 창 필요 탑 → shots/v13_*.png
import sys, os, asyncio
from playwright.async_api import async_playwright
ROOT = os.path.abspath(os.path.join(os.path.dirname(os.path.abspath(__file__)), '..'))
OUT = ROOT + '/shots/'
async def main():
    W, H = 360, 742
    if len(sys.argv) > 1 and 'x' in sys.argv[1]: W, H = map(int, sys.argv[1].split('x'))
    lang = sys.argv[2] if len(sys.argv) > 2 else ''
    pre = 'v13_' + (lang + '_' if lang else '')
    async with async_playwright() as p:
        b = await p.chromium.launch()
        ctx = await b.new_context(viewport={'width': W, 'height': H}, device_scale_factor=2, is_mobile=True, has_touch=True)
        pg = await ctx.new_page()
        errs = []
        pg.on('pageerror', lambda e: errs.append('PAGE: ' + str(e)))
        pg.on('console', lambda m: errs.append(m.type + ': ' + m.text) if m.type == 'error' else None)
        await pg.goto('file://' + ROOT + '/out/index.html' + ('?lang=' + lang if lang else ''))
        await pg.wait_for_timeout(1200)
        async def ev(js): return await pg.evaluate(js)
        async def shot(n, w=500): await pg.wait_for_timeout(w); await pg.screenshot(path=OUT + pre + n + '.png')
        await ev("__td.fakeMaps(20); __td.unlockAll(); var s=__td.SAVE(); s.prog=13; s.gems=420; s.tickets=3; s.towers.frost={n:3,star:2}; s.towers.cannon={n:2,star:1}; __td.MAPS[13].need=['frost','bell','cannon'];")
        await pg.touchscreen.tap(W / 2, H / 2); await pg.wait_for_timeout(1200)
        await shot('map_p1', 300)
        await ev("var w=document.getElementById('mapWrap'); w.scrollTop=0;")
        await shot('map_p2top', 400)
        await ev("var w=document.getElementById('mapWrap'), n=document.querySelectorAll('.node')[11]; w.scrollTop += n.getBoundingClientRect().top - w.clientHeight*0.6;")
        await shot('map_p2join', 400)
        await ev("__meta.stage(13)"); await shot('stage_need')
        await ev("__gc.open()"); await shot('gacha')
        await ev("__gc.prob()"); await shot('prob')
        await ev("__gc.res(1)"); await shot('res1_mid', 350); await shot('res1', 1300)
        await ev("__gc.res(10)"); await shot('res10_mid', 700); await ev("__gc.skip()"); await shot('res10', 900)
        await ev("__meta.close(); __td.start(0,'normal')"); await pg.wait_for_timeout(1200)
        await ev("__gc.fakePick(true)"); await shot('pick_start', 700)
        await ev("document.querySelector('.pkc').click()"); await pg.wait_for_timeout(900)
        await ev("__gc.fakePick(false, true)"); await shot('pick_combo', 700)
        await ev("document.querySelectorAll('.pkc')[2].click()"); await pg.wait_for_timeout(900)
        await ev("__gc.fakeCombo()"); await shot('combo', 350)
        await ev("__gc.build('archer')"); await shot('build', 500)
        print('오류:', errs[:8])
        await b.close()
asyncio.run(main())
