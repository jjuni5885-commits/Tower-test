  M.forEach(function (m, i) { if (o.sea < 0 && m.sea && m.sea.length) o.sea = i; if (o.air < 0 && m.air && m.air.length) o.air = i; });
  if (o.sea < 0) o.sea = 0; if (o.air < 0) o.air = 0; return o; })()"""

SCENES = {}
def scene(name, mapk): 
    def d(f): SCENES[name] = (mapk, f); return f
    return d

async def freeze(pg):
    # 일시 정지 상태로 두고(창은 숨김) adv로 정해진 프레임만큼만 돌린다 → 연출을 원하는 순간에 찍을 수 있음
    await pg.evaluate("window.__appPause(); document.getElementById('sModal').classList.add('hide')")
async def adv(pg, n=1):
    await pg.evaluate("__td.prof(%d)" % n); await pg.wait_for_timeout(60)
async def settle(pg, steps=3, ms=0):
    await freeze(pg); await adv(pg, steps)

async def run(names, sizes):
    async with async_playwright() as p:
        b = await p.chromium.launch()
        for sz in sizes:
            W, H = map(int, sz.split('x'))
            for nm in names:
                if nm == 'perf': continue
                mapk, fn = SCENES[nm]
                ctx = await b.new_context(viewport={'width': W, 'height': H}, device_scale_factor=2, is_mobile=True, has_touch=True)
                pg = await ctx.new_page(); errs = []
                pg.on('pageerror', lambda e: errs.append('PAGE: ' + str(e)))
                pg.on('console', lambda m: errs.append(m.type + ': ' + m.text) if m.type == 'error' else None)
                await pg.goto('file://' + ROOT + '/out/index.html'); await pg.wait_for_timeout(900)
                picks = await pg.evaluate(PICK)
                await pg.evaluate("__td.SAVE().tut.done=1; __td.SAVE().set.music=false; Object.keys(__td.EN).forEach(function(k){ __td.SAVE().seen[k]=1; }); __td.unlockAll(); __td.ownAll(); __td.start(%d,'normal')" % picks[mapk]); await pg.wait_for_timeout(900)
                await pg.evaluate("__td.closeModal()")
                await pg.evaluate(HELP)
                await pg.evaluate("(function(){ var G = __td.G(); G.pickWait = false; G.autoWave = false; G.nextT = 0; })()")
                async def shot(tag):
                    path = OUT + 'ru_%s_%s.png' % (tag, sz); await pg.screenshot(path=path); print('  ', path)
                print(nm, sz)
                await fn(pg, shot)
                st = await pg.evaluate("(function(){ var s = __td.st(); return { err: s.err, parts: __td.pc().n, fx: __ru.RU.fx.length }; })()")
                print('  상태', st, '콘솔 오류', len(errs), errs[:4])
                await ctx.close()
            if 'perf' in names:
                ctx = await b.new_context(viewport={'width': W, 'height': H}, device_scale_factor=2, is_mobile=True, has_touch=True)
                pg = await ctx.new_page(); await pg.goto('file://' + ROOT + '/out/index.html'); await pg.wait_for_timeout(900)
                picks = await pg.evaluate(PICK)
