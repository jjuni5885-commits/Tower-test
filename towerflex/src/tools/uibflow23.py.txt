OUT = ROOT + '/shots/'
RES = []
def ok(name, cond, info=''):
    RES.append((name, bool(cond)))
    print(('통과 ' if cond else '실패 ') + name + ('  ' + str(info) if info != '' else ''))

  var T = __uib.TOWERS, h = __uib.has(), out = { noArt: [], short: [], fuse: h.fuse, adm: h.adm };

  });
  return out;
})()"""

            await pg.wait_for_timeout(wait); await pg.screenshot(path=OUT + 'uib_%s_%s.png' % (tag, n))
