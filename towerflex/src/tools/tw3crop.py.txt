# 그림판 일부를 잘라 키워 보기: python3 tools/tw3crop.py <png> <x0> <y0> <x1> <y1> [배율] → shots/_crop.png (0~1 비율 좌표)
import sys
from PIL import Image
im = Image.open(sys.argv[1]); W, H = im.size
x0, y0, x1, y1 = [float(v) for v in sys.argv[2:6]]; k = float(sys.argv[6]) if len(sys.argv) > 6 else 1
c = im.crop((int(x0 * W), int(y0 * H), int(x1 * W), int(y1 * H)))
if k != 1: c = c.resize((int(c.width * k), int(c.height * k)), Image.NEAREST if k >= 2 else Image.LANCZOS)
out = sys.argv[7] if len(sys.argv) > 7 else sys.argv[1].rsplit('/', 1)[0] + '/_crop.png'
c.save(out); print(out, c.size)
