async def main():
    mode = sys.argv[1] if len(sys.argv) > 1 else 'all'
    os.makedirs(OUT, exist_ok=True)
    async with async_playwright() as p:
        b = await p.chromium.launch(args=['--autoplay-policy=no-user-gesture-required'])
        errs = []
        if mode == 'battle':
            # 10x17 격자, 물 칸이 있는 시험 판(c1s04)에서 36px 칸으로 확인: python3 tools/t4shot.py battle [판 번호] [가로x세로]
            W, H = 360, 640
            if len(sys.argv) > 3 and 'x' in sys.argv[3]: W, H = map(int, sys.argv[3].split('x'))
            ctx = await b.new_context(viewport={'width': W, 'height': H}, device_scale_factor=3, is_mobile=True, has_touch=True)
            pg = await ctx.new_page()
            pg.on('console', lambda m: errs.append(m.type + ': ' + m.text) if m.type in ('error',) else None)
            pg.on('pageerror', lambda e: errs.append('PAGE: ' + str(e)))
            await pg.goto('file://' + ROOT + '/out/index.html')
            await pg.wait_for_timeout(1200)
            mi = int(sys.argv[2]) if len(sys.argv) > 2 else await pg.evaluate("(function(){var s=0; __td.MAPS.forEach(function(m,i){ if(!s&&m.sea&&m.sea.length) s=i; }); return s;})()")
            await pg.evaluate("__td.SAVE().tut.done=1; __td.unlockAll(); __td.ownAll(); __td.start(%d,'normal')" % mi)
            await pg.wait_for_timeout(1500)
            r = await pg.evaluate("""(function(){ var g = __td.G(); g.gold = 999999; var out = [], v = __td.view();
              function put(k, cells, lv) { for (var i = 0; i < cells.length; i++) { var t = __td.build(k, cells[i][0], cells[i][1]); if (t) { t.lvl = lv; t.cd = 99; out.push(k + '@' + cells[i] + ' L' + lv); return t; } } out.push(k + ' 못 지음'); return null; }
              var water = [], land = []; for (var r = 1; r < g.rows - 1; r++) for (var c = 0; c < g.cols; c++) { var kd = g.kind[r * g.cols + c]; if (kd === 'water') water.push([c, r]); else if (kd === 'g') land.push([c, r]); }
              function far(list, used, d) { return list.filter(function (p) { return used.every(function (q) { return Math.abs(p[0] - q[0]) + Math.abs(p[1] - q[1]) >= d; }); }); }
              var used = [];
              [['fz_carrier', 1], ['fz_dragonship', 2], ['fz_carrier', 3], ['fz_dragonship', 1]].forEach(function (k) { var t = put(k[0], far(water, used, 3), k[1]); if (t) used.push([t.c, t.r]); });
              [['fz_aegis', 1], ['fz_skybow', 2], ['fz_fortress', 3], ['fz_storm', 1], ['fz_storm', 3], ['fz_frostfire', 2], ['fz_arsenal', 3], ['omega', 2], ['dragon', 1], ['dragon', 3], ['archer', 3], ['cannon', 3]].forEach(function (k) { var t = put(k[0], far(land, used, 3), k[1]); if (t) used.push([t.c, t.r]); });
              return { built: out, cols: g.cols, rows: g.rows, cellCss: +(v.S / v.dpr).toFixed(1), id: g.M.id }; })()""")
            print(r)
            await pg.evaluate("var m = document.getElementById('sModal'); if (m) m.classList.add('hide');")
            await pg.wait_for_timeout(900)
            await pg.screenshot(path=OUT + 't4_battle0.png')
            await pg.evaluate("(function(){ var g = __td.G(); g.towers.forEach(function (t, i) { t.aim = i % 2 ? 2.6 : 0.4; t.fireT = 0.3; t.spin = 5; t.busy = 0.8; t.shots = i; t.cd = 99; if (i % 5 === 4) t.out = 1; }); })()")
            await pg.wait_for_timeout(50)
            await pg.screenshot(path=OUT + 't4_battle1.png')
            await pg.wait_for_timeout(2500)
            await pg.screenshot(path=OUT + 't4_battle2.png')
            st = await pg.evaluate("(function(){ return { fps: Math.round(__td.fps()), err: __td.st().err || 0 }; })()")
            print('state', st)
        else:
            tags = ''.join('<script src="file://%s/src/%s"></script>\n' % (ROOT, f) for f in ['util.js', 'data.js', 'data_towers2.js', 'art_core.js', 'art_tower.js', 'art_tower4.js'])
            hp = '/tmp/t4_harness.html'
            open(hp, 'w').write(HARN.replace('%s', tags, 1).replace('%%', '%'))
            pg = await b.new_page(viewport={'width': 1600, 'height': 1600}, device_scale_factor=1)
            pg.on('console', lambda m: errs.append(m.type + ': ' + m.text) if m.type in ('error', 'warning') else None)
            pg.on('pageerror', lambda e: errs.append('PAGE: ' + str(e)))
            await pg.goto('file://' + hp)
            await pg.wait_for_timeout(300)
            async def snap(name):
                await pg.locator('#cv').screenshot(path=OUT + name)
                print('saved', OUT + name)
            if mode == 'one':
                for k in sys.argv[2].split(','):
                    e = await pg.evaluate("h4One('%s')" % k); await snap('t4_one_%s.png' % k)
                    if e: print('ERR', e)
            elif mode == 'all':
                ks = sys.argv[2].split(',') if len(sys.argv) > 2 else KEYS
                for i in range(0, len(ks), 4):
                    e = await pg.evaluate("h4All(%s, 110, 6)" % json.dumps(ks[i:i + 4])); await snap('t4_all_%d.png' % (i // 4))
                    if e: print('ERR', e)
            elif mode == 'small':
                e = await pg.evaluate("h4Small(%s)" % json.dumps(KEYS)); await snap('t4_small.png'); print('ERR', e) if e else None
            elif mode == 'cmp':
                e = await pg.evaluate("h4Cmp(%s)" % json.dumps(KEYS)); await snap('t4_cmp.png'); print('ERR', e) if e else None
            elif mode == 'dragon':
                e = await pg.evaluate("h4Dragon()"); await snap('t4_dragon.png'); print('ERR', e) if e else None
            elif mode == 'perf':
                ks = sys.argv[2].split(',') if len(sys.argv) > 2 else KEYS
                await pg.evaluate("h4Perf(%s, 96)" % json.dumps(ks))
                r = await pg.evaluate("h4Perf(%s, 108)" % json.dumps(ks)); print(json.dumps(r, indent=0))
        print('errors', len(errs))
        for e in errs[:20]: print(e)
        await b.close()
asyncio.run(main())
