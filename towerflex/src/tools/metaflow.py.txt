# 수익화([BM])와 재방문 기능 시험: python3 tools/metaflow.py [가로x세로]
# 손가락 탭으로: 전투 처치 → 처음 처치 기록과 일일 임무 집계 → 도감 보석 +5(한 번만) → 임무 받기 → 상점(모자람, 사기, 장착, 화살 스킨)
#   → 새로 고침 뒤 유지 → 광고 자리 5초 보상, 중간에 닫으면 보상 없음, 하루 5번 제한 → 날짜 바뀜(Date 흉내) → 패배 → 광고 이어하기 → 다시 지면 결과 창
import sys, os, asyncio
from playwright.async_api import async_playwright
ROOT = os.path.abspath(os.path.join(os.path.dirname(os.path.abspath(__file__)), '..'))
OUT = ROOT + '/shots/'
RES = []
def ok(name, cond, info=''):
    RES.append((name, bool(cond)))
    print(('통과 ' if cond else '실패 ') + name + ('  ' + str(info) if info != '' else ''))
# 기기 날짜 흉내: localStorage '__dayShift'(일) 만큼 Date를 옮김 (새로 고침해도 유지)
DATE_SHIM = r"""(() => { const RD = Date; const sh = () => { try { return (parseFloat(localStorage.getItem('__dayShift') || '0') || 0) * 864e5; } catch (e) { return 0; } };
  function FD(...a) { if (!new.target) return new RD(RD.now() + sh()).toString(); return a.length ? new RD(...a) : new RD(RD.now() + sh()); }
  FD.prototype = RD.prototype; FD.now = () => RD.now() + sh(); FD.parse = RD.parse; FD.UTC = RD.UTC; window.Date = FD; })();"""

async def main():
    W, H = 360, 640
    if len(sys.argv) > 1 and 'x' in sys.argv[1]: W, H = map(int, sys.argv[1].split('x'))
    async with async_playwright() as p:
        b = await p.chromium.launch()
        ctx = await b.new_context(viewport={'width': W, 'height': H}, device_scale_factor=3, is_mobile=True, has_touch=True)
        await ctx.add_init_script(DATE_SHIM)
        pg = await ctx.new_page()
        errs = []
        pg.on('pageerror', lambda e: errs.append('PAGE: ' + str(e)))
        pg.on('console', lambda m: errs.append(m.type + ': ' + m.text) if m.type == 'error' else None)
        await pg.goto('file://' + ROOT + '/out/index.html')
        await pg.wait_for_timeout(1300)
        async def ev(js): return await pg.evaluate(js)
        async def tap_el(sel, wait=450, idx=0):
            r = await ev("""(function(){var e=document.querySelectorAll(%r)[%d]; if(!e) return null; var b=e.getBoundingClientRect();
              if (b.top < 0 || b.bottom > innerHeight) { e.scrollIntoView({block:'center'}); b=e.getBoundingClientRect(); }
              var x=b.left+b.width/2, y=b.top+b.height/2, t=document.elementFromPoint(x,y); return {x:x,y:y,hit:!!t&&(e===t||e.contains(t))};})()""" % (sel, idx))
            if not r: ok('탭 대상 있음 ' + sel, False); return None
            if not r['hit']: ok('탭 가려지지 않음 ' + sel, False)
            await pg.touchscreen.tap(r['x'], r['y']); await pg.wait_for_timeout(wait)
            return r

                await tap_el('#bOk', 350); n += 1

            t = 0

                if await ev("__td.st().kind==='result'"): return True
                await pg.wait_for_timeout(250); t += 250
