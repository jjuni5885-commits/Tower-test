# 화면 확인: python3 tools/shot.py [모드] ...
import sys, asyncio, json
from playwright.async_api import async_playwright
import os
ROOT = os.path.abspath(os.path.join(os.path.dirname(os.path.abspath(__file__)), '..'))
OUT = ROOT + '/shots/'
async def main():
    mode = sys.argv[1] if len(sys.argv) > 1 else 'basic'
    W, H = 390, 844
    if len(sys.argv) > 3 and 'x' in sys.argv[3]: W, H = map(int, sys.argv[3].split('x'))
    async with async_playwright() as p:
        b = await p.chromium.launch(args=['--autoplay-policy=no-user-gesture-required'])
        pg = await b.new_page(viewport={'width': W, 'height': H}, device_scale_factor=2)
        errs = []
        pg.on('console', lambda m: errs.append(m.type + ': ' + m.text) if m.type in ('error', 'warning') else None)
        pg.on('pageerror', lambda e: errs.append('PAGE: ' + str(e)))
        await pg.goto('file://' + ROOT + '/out/index.html')
        await pg.wait_for_timeout(1500)
        if mode == 'basic':
            await pg.screenshot(path=OUT + 'title.png')
            await pg.mouse.click(W / 2, H / 2)
            await pg.wait_for_timeout(1500)
            await pg.screenshot(path=OUT + 'map.png')
            await pg.evaluate("__td.unlockAll()")
            await pg.evaluate("document.querySelectorAll('.node')[0].click()")
            await pg.wait_for_timeout(800)
            await pg.screenshot(path=OUT + 'stage.png')
        elif mode == 'sheet':
            import base64
            for k in ['towers', 'enemies']:
                d = await pg.evaluate("__td.sheet('%s', %d)" % (k, 110 if k == 'towers' else 130))
                await pg.wait_for_timeout(300)
                d = await pg.evaluate("__td.sheet('%s', %d)" % (k, 110 if k == 'towers' else 130))
                open(OUT + 'sheet_%s.png' % k, 'wb').write(base64.b64decode(d.split(',')[1]))
        elif mode == 'battle':
            mi = int(sys.argv[2]) if len(sys.argv) > 2 else 0
            await pg.mouse.click(W / 2, H / 2)
            await pg.wait_for_timeout(500)
            await pg.evaluate("__td.unlockAll(); Object.keys(ENEMIES_ALL()).forEach(function(k){ __td.SAVE().seen[k]=1; })".replace('ENEMIES_ALL()', '__td.EN'))
            await pg.evaluate("__td.start(%d, 'normal')" % mi)
            await pg.wait_for_timeout(2500)
            await pg.screenshot(path=OUT + 'b%d_0.png' % mi)
            await pg.evaluate("__td.bot(1); __td.speed(3)")
            for k in range(int(sys.argv[4]) if len(sys.argv) > 4 else 3):
                await pg.wait_for_timeout(9000)
                await pg.screenshot(path=OUT + 'b%d_%d.png' % (mi, k + 1))
            st = await pg.evaluate("(function(){var g=__td.G(); return {wave:g.wave, lives:g.lives, gold:g.gold, towers:g.towers.length, en:g.enemies.length, fps:__td.fps()};})()")
            print('state', json.dumps(st))
        print('errors', len(errs))
        for e in errs[:20]: print(e)
        await b.close()
if not (len(sys.argv) > 1 and sys.argv[1] in ('gallery', 'ui', 'flows', 'perf', 'misc')):
    asyncio.run(main())

async def gallery(maps, secs, tag=''):
    async with async_playwright() as p:
        b = await p.chromium.launch(args=['--autoplay-policy=no-user-gesture-required'])
        pg = await b.new_page(viewport={'width': 390, 'height': 844}, device_scale_factor=2)
        errs = []
        pg.on('pageerror', lambda e: errs.append('PAGE: ' + str(e)))
        pg.on('console', lambda m: errs.append(m.type + ': ' + m.text) if m.type == 'error' else None)
        await pg.goto('file://' + ROOT + '/out/index.html')
        await pg.wait_for_timeout(1200)
        await pg.mouse.click(195, 400)
        await pg.wait_for_timeout(400)
        await pg.evaluate("__td.unlockAll(); Object.keys(__td.EN).forEach(function(k){ __td.SAVE().seen[k]=1; })")
        for mi in maps:
            await pg.evaluate("__td.start(%d, 'normal')" % mi)
            await pg.wait_for_timeout(1500)
            await pg.evaluate("__td.bot(1); __td.speed(3)")
            await pg.wait_for_timeout(secs * 1000)
            await pg.screenshot(path=OUT + 'g%d%s.png' % (mi, tag))
            st = await pg.evaluate("(function(){var g=__td.G(); return [g.wave, g.lives, g.towers.length, g.enemies.length, Math.round(__td.fps())];})()")
            print(mi, st)
        print('errors', len(errs)); [print(e) for e in errs[:10]]
        await b.close()

async def uitest():
    async with async_playwright() as p:
        b = await p.chromium.launch()
        UW, UH = (map(int, sys.argv[2].split('x')) if len(sys.argv) > 2 else (390, 844))
        pg = await b.new_page(viewport={'width': UW, 'height': UH}, device_scale_factor=2)
        errs = []
        pg.on('pageerror', lambda e: errs.append('PAGE: ' + str(e)))
        pg.on('console', lambda m: errs.append(m.type + ': ' + m.text) if m.type == 'error' else None)
        await pg.goto('file://' + ROOT + '/out/index.html')
        await pg.wait_for_timeout(1200)
        await pg.mouse.click(UW // 2, UH // 2); await pg.wait_for_timeout(300)
        await pg.evaluate("__td.unlockAll(); Object.keys(__td.EN).forEach(function(k){ __td.SAVE().seen[k]=1; }); __td.SAVE().tut.done=1")
        await pg.evaluate("__td.start(4, 'normal')")
        await pg.wait_for_timeout(1500)
        await pg.evaluate("__td.G().gold = 5000")
        # 빈 칸 화면 좌표 구하기
        async def cellXY(c, r):
            return await pg.evaluate("(function(){var v=RDV(); return [(v.ox + (%d + 0.5) * v.S) / v.dpr, (v.oy + (%d + 0.5) * v.S) / v.dpr];})()".replace('RDV()', '__td.view()') % (c, r))
        cr = await pg.evaluate("(function(){var g=__td.G(); for (var r=3;r<14;r++) for (var c=0;c<8;c++) { if (g.kind[r*8+c]==='g' && g.obsAt[r*8+c]<0) { var near=false; for (var dr=-1;dr<=1;dr++) for (var dc=-1;dc<=1;dc++) { var rr=r+dr, cc=c+dc; if (rr>=0&&rr<14&&cc>=0&&cc<8&&g.kind[rr*8+cc]==='p') near=true; } if (near) return [c,r]; } } })()")
        x, y = await cellXY(cr[0], cr[1])
        await pg.mouse.click(x, y); await pg.wait_for_timeout(400)
        await pg.screenshot(path=OUT + 'ui_build0.png')
        await pg.click('.tc[data-k="archer"]'); await pg.wait_for_timeout(300)
        await pg.screenshot(path=OUT + 'ui_build1.png')
        await pg.click('.tc[data-k="archer"]'); await pg.wait_for_timeout(400)
        await pg.mouse.click(x, y); await pg.wait_for_timeout(300)
        for i in range(2):
            await pg.click('.uc'); await pg.wait_for_timeout(150); await pg.click('.uc'); await pg.wait_for_timeout(300)
        await pg.screenshot(path=OUT + 'ui_branch.png')
        await pg.click('.uc[data-br="1"]'); await pg.wait_for_timeout(300)
        await pg.screenshot(path=OUT + 'ui_branch_sel.png')
        await pg.click('.uc[data-br="1"]'); await pg.wait_for_timeout(500)
        await pg.screenshot(path=OUT + 'ui_tower4.png')
        await pg.click('#bPause'); await pg.wait_for_timeout(300)
        await pg.screenshot(path=OUT + 'ui_pause.png')
        print('errors', len(errs)); [print(e) for e in errs[:10]]
        await b.close()
if len(sys.argv) > 1 and sys.argv[1] == 'ui':
    asyncio.run(uitest())
