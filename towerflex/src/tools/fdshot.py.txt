# v2.0 기반 확인용 전투 캡처: python3 tools/fdshot.py [가로x세로 ...]  → shots/fd_<크기>_<판 id>.png (땅 판, 물 판, 항로 판)
import sys, asyncio, os
from playwright.async_api import async_playwright
ROOT = os.path.abspath(os.path.join(os.path.dirname(os.path.abspath(__file__)), '..'))
OUT = ROOT + '/shots/'
async def main():
    sizes = [a for a in sys.argv[1:] if 'x' in a] or ['360x640', '360x780', '412x915']
    os.makedirs(OUT, exist_ok=True)
    async with async_playwright() as p:
        b = await p.chromium.launch()
        for sz in sizes:
            W, H = map(int, sz.split('x'))
            ctx = await b.new_context(viewport={'width': W, 'height': H}, device_scale_factor=2, is_mobile=True, has_touch=True)
            pg = await ctx.new_page(); errs = []
            pg.on('pageerror', lambda e: errs.append('PAGE: ' + str(e)))
            pg.on('console', lambda m: errs.append(m.type + ': ' + m.text) if m.type == 'error' else None)
            await pg.goto('file://' + ROOT + '/out/index.html'); await pg.wait_for_timeout(1200)
            n = await pg.evaluate("__td.MAPS.length")
            picks = await pg.evaluate("(function(){var M=__td.MAPS, o=[0], s=-1, a=-1; M.forEach(function(m,i){ if(s<0&&m.sea&&m.sea.length) s=i; if(a<0&&m.air&&m.air.length) a=i; }); if(s>=0) o.push(s); if(a>=0) o.push(a); return o;})()")
            for mi in picks:
                await pg.evaluate("__td.SAVE().tut.done=1; __td.unlockAll(); __td.ownAll(); __td.start(%d,'normal')" % mi); await pg.wait_for_timeout(1200)
                info = await pg.evaluate("""(function(){ var g=__td.G(); g.gold=5000; __td.bot(1); for (var i=0;i<60*75 && !g.over;i++) __td.step(1);
                  var v=__td.view(); return {id:g.M.id, cols:g.cols, rows:g.rows, cellCss: +(v.S/v.dpr).toFixed(1), gridW: Math.round(v.S*g.cols/v.dpr), gridH: Math.round(v.S*g.rows/v.dpr), top: v.top, bot: v.bot, wave:g.wave, towers:g.towers.length, enemies:g.enemies.length, sea:g.enemies.filter(function(e){return e.dom==='s';}).length, air:g.enemies.filter(function(e){return e.lane;}).length, lives:g.lives}; })()""")
                await pg.evaluate("document.getElementById('sModal').classList.add('hide')"); await pg.wait_for_timeout(700)
                path = OUT + 'fd_%s_%s.png' % (sz, info['id'])
                await pg.screenshot(path=path)
                print(sz, info, path)
            print(sz, '콘솔 오류:', errs[:5])
            await ctx.close()
        await b.close()
asyncio.run(main())
