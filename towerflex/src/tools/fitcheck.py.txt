CHECK = r"""(function(arg){

  var lim0 = edge ? 0 : sat, lim1 = edge ? H : H - sab;

  if (M.scrollHeight > M.clientHeight + 1) fails.push('창 스크롤 ' + M.scrollHeight + '>' + M.clientHeight);
  var mcs = getComputedStyle(M);
  if (M.scrollWidth > M.clientWidth + 1 && !/(hidden|clip)/.test(mcs.overflowX)) fails.push('창 가로 넘침 ' + M.scrollWidth + '>' + M.clientWidth);
  var holder = M.parentNode; if (holder && holder.scrollHeight > holder.clientHeight + 1) fails.push('창 바탕 스크롤');
  var nIn = 0, nBtn = 0;
  M.querySelectorAll('*').forEach(function(e){
    var cs = getComputedStyle(e); if (cs.display === 'none' || cs.visibility === 'hidden' || e.offsetParent === null && cs.position !== 'fixed') return;
    var isAllow = allow.some(function(a){ return e.matches(a); });
    var inAllow = allow.some(function(a){ var c = e.parentElement && e.parentElement.closest(a); return !!c && M.contains(c); });
    if (/(auto|scroll)/.test(cs.overflowY) && e.scrollHeight > e.clientHeight + 1 && !isAllow) fails.push('안쪽 스크롤 ' + desc(e));
    if (inAllow) return;
    var b = e.getBoundingClientRect(); if (b.width === 0 && b.height === 0) return;

    if (b.top < r.top - 1 || b.bottom > r.bottom + 1 || b.left < r.left - 1 || b.right > r.right + 1) fails.push('카드 밖 ' + desc(e) + ' ' + [b.top, b.bottom].map(Math.round).join('/'));
  });
  M.querySelectorAll('button').forEach(function(b){
    var bb = b.getBoundingClientRect(); if (!bb.width || !bb.height) return;

    var x = bb.left + bb.width / 2, y = bb.top + bb.height / 2, t = document.elementFromPoint(x, y);

async def run_size(p, spec):
    safe = spec.endswith('s'); W, H = map(int, spec.rstrip('s').split('x'))
    b = await p.chromium.launch()
    ctx = await b.new_context(viewport={'width': W, 'height': H}, device_scale_factor=3, is_mobile=True, has_touch=True)
    pg = await ctx.new_page()
    errs = []
    pg.on('pageerror', lambda e: errs.append('PAGE: ' + str(e)))
    pg.on('console', lambda m: errs.append(m.type + ': ' + m.text) if m.type == 'error' else None)
    if safe:
        cdp = await ctx.new_cdp_session(pg)
        await cdp.send('Emulation.setSafeAreaInsetsOverride', {'insets': {'top': 30, 'bottom': 24, 'left': 0, 'right': 0}})
    await pg.goto('file://' + ROOT + '/out/index.html')
    await pg.wait_for_timeout(1300)
    res = []
    async def ev(js, arg=None): return await pg.evaluate(js, arg) if arg is not None else await pg.evaluate(js)
    async def check(name, sel, shot=None, wait=420, edge=False):
        await pg.wait_for_timeout(wait)
        r = await ev(CHECK, [sel, ALLOW, edge])
        ok = not r['fails']
        res.append((name, ok, r))
        if not ok: print('  실패 %-14s %s' % (name, r['fails'][:4]))
        if shot and spec in SHOT_SIZES: await pg.screenshot(path=OUT + 'fit_%s_%s.png' % (spec, shot))
        return r
    async def tap_sel(sel, idx=0, wait=450):
        r = await ev("""(function(a){var e=document.querySelectorAll(a[0])[a[1]]; if(!e) return null; var b=e.getBoundingClientRect(); var x=b.left+b.width/2, y=b.top+b.height/2, t=document.elementFromPoint(x,y); return {x:x,y:y,hit:!!t&&(t===e||e.contains(t))};})""", [sel, idx])
        if not r: res.append(('탭 ' + sel, False, {'fails': ['없음']})); print('  없음', sel); return None
        if not r['hit']: res.append(('탭 ' + sel, False, {'fails': ['가려짐']})); print('  가려짐', sel)
        await pg.touchscreen.tap(r['x'], r['y']); await pg.wait_for_timeout(wait)
        return r
    async def wait_kind(kind, ms=6000):
        # 창이 바뀔 때까지 기다림(시험 기계가 느리면 탭 뒤 창이 늦게 뜬다)
        t = 0
        while t < ms:
            if await ev("__td.st().kind") == kind: await pg.wait_for_timeout(350); return True
            await pg.wait_for_timeout(200); t += 200
        res.append(('창 ' + kind, False, {'fails': ['안 뜸']})); print('  안 뜸', kind); return False
    async def tap_node(i):
        # 지도를 굴려 노드를 화면 가운데쯤에 두고 손가락 탭 (touchflow.py 방식)
        r = await ev("""(function(i){var n=document.querySelectorAll('.node')[i], w=document.getElementById('mapWrap'); var b=n.getBoundingClientRect(); w.scrollTop += b.top - w.clientHeight*0.45; return true;})""", i)
        await pg.wait_for_timeout(250)
        r = await ev("""(function(i){var n=document.querySelectorAll('.node')[i].querySelector('.nb'); var b=n.getBoundingClientRect(); var x=b.left+b.width/2, y=b.top+b.height/2, t=document.elementFromPoint(x,y); return {x:x,y:y,hit:!!t&&(t===n||n.contains(t)||t.closest('.node')===n.parentNode)};})""", i)
        if not r['hit']: res.append(('노드 %d 가려짐' % (i + 1), False, {'fails': ['가려짐']}))
        await pg.touchscreen.tap(r['x'], r['y']); await pg.wait_for_timeout(300)
        await wait_kind('stage')
    async def close_modal():
        if await ev("!document.getElementById('sModal').classList.contains('hide')"):
            sel = await ev("""(function(){var m=document.getElementById('modal'); var b=m.querySelector('#bClose')||m.querySelector('.xbtn')||m.querySelector('#rDone'); return b ? (b.id ? '#'+b.id : '.xbtn') : null;})()""")
            if sel: await tap_sel('#modal ' + sel if not sel.startswith('#') else sel, 0, 450)
            else: await ev("__meta.close()")

    # 준비: 모든 지역 열기, 도감 보상 하나 받을 수 있게, 보석 조금
    await ev("__ch.fake(12); __td.unlockAll(); var s=__td.SAVE(); s.seen={goblin:1,orc:1,wolf:1,knight:1,bat:1,stalker:1}; s.slain={goblin:1,orc:1}; s.codexGot={goblin:1}; s.fused={fz_skybow:1}; s.towers.frost={n:1,star:2}; s.towers.warship={n:1,star:0}; s.towers.sniper={n:2,star:1}; __meta.gems(75); __td.MAPS[1].need=['archer','frost','bell'];")

    await check('챕터 고르기', '#chSel', 'chsel', 300, True)

    await ev("var s=__td.SAVE(); s.from='lastarrow.v4'; delete s.fromSeen; __ch.sel();")

    await tap_sel('#mgOk', 0, 400)
    # 챕터 지도 4장: 가짜 보너스 칸까지 넣어 노드가 막대에 가리지 않고 스크롤 없이 들어가는지
    for ch in [4, 3, 2, 1]:
        await ev("__ch.map(%d)" % ch)
        await check('지도 %d장' % ch, '#nodes', 'chmap%d' % ch, 350, True)

async def main():
    sizes = SIZES
    args = [a for a in sys.argv[1:] if not a.startswith('--')]
    if args: sizes = args[0].split(',')
    total = 0; failed = 0; allerr = []
    async with async_playwright() as p:
        for s in sizes:
            print('[%s]' % s)
            res, errs = await run_size(p, s)
            n = len(res); f = [x[0] for x in res if not x[1]]
            total += n; failed += len(f)
            print('  %d/%d 통과%s' % (n - len(f), n, ('  실패: ' + ', '.join(f)) if f else ''), ' 콘솔 오류 %d' % len(errs))
            allerr += errs
    print('전체: %d/%d 통과, 콘솔 오류 %d' % (total - failed, total, len(allerr)))
    for e in allerr[:10]: print('  ', e)
asyncio.run(main())
