    assert a in h
    h = h.replace(a, "var top = !!(EN_DRAW[e.k] && EN_DRAW[e.k].top);")
    open(OUT + '/test.html', 'w', encoding='utf8').write(h)

def save(d, name):
    open(OUT + '/' + name, 'wb').write(base64.b64decode(d.split(',')[1])); print('saved', OUT + '/' + name)
async def main():
    modes = (sys.argv[1] if len(sys.argv) > 1 else 'big,small,frames,rot,line').split(',')
    keys = sys.argv[2].split(',') if len(sys.argv) > 2 else K3
    mk_test()
    async with async_playwright() as p:
        b = await p.chromium.launch()
        pg = await b.new_page(viewport={'width': 360, 'height': 640}, device_scale_factor=2)
        errs = []
        pg.on('pageerror', lambda e: errs.append('PAGE ' + str(e)))
        pg.on('console', lambda m: errs.append(m.type + ': ' + m.text) if m.type in ('error', 'warning') else None)
        await pg.goto('file://' + OUT + '/test.html'); await pg.wait_for_timeout(1200)
        await pg.evaluate(JS)
        tops = await pg.evaluate("(%s).filter(function(k){ return __ED[k] && __ED[k].top; })" % json.dumps(keys))
        kj = json.dumps(keys)
        if 'big' in modes: save(await pg.evaluate("__B.icons(%s, 138, 5)" % kj), 'big.png')
        if 'huge' in modes: save(await pg.evaluate("__B.icons(%s, 300, 4)" % kj), 'huge.png')
        if 'small' in modes: save(await pg.evaluate("__B.icons(%s, 48, 10)" % kj), 'small.png')
        if 'frames' in modes:
            half = (len(keys) + 1) // 2
            save(await pg.evaluate("__B.frames(%s, 36, 2)" % json.dumps(keys[:half])), 'frames_a.png')
            if keys[half:]: save(await pg.evaluate("__B.frames(%s, 36, 2)" % json.dumps(keys[half:])), 'frames_b.png')
        if 'rot' in modes and tops: save(await pg.evaluate("__B.rot(%s, 36, 2)" % json.dumps(tops)), 'rot.png')
        if 'line' in modes:
            save(await pg.evaluate("__B.line(%s, 36, 2, ['#70757c', '#d9c08c', '#3f84b5'])" % json.dumps([k for k in keys if k != 'submarine_sub'] + ['goblin', 'knight', 'wolf', 'wagon', 'golemL', 'bat', 'wyvern', 'grumsh'])), 'line.png')
        if 'perf' in modes:
            r = await pg.evaluate("__B.perf(%s)" % kj)
            for k, v in r.items(): print(k, v, 'SLOW' if max(float(q.split('/')[1]) for q in v) > 30 else '')  # 크기 36, 41, 67, 97에서 최소/중앙값(ms), 7번 반복
        if 'stress' in modes:  # 아주 작은 크기부터 큰 크기까지 예외 없이 만들어지는지
            r = await pg.evaluate("(function(keys){ var bad = []; keys.forEach(function(k){ [8, 12, 16, 24, 36, 45, 70, 130, 200].forEach(function(S){ try { var fr = __EF(k, S + 0.2); if (!fr.fr.length || fr.fr.length !== fr.wh.length || fr.fr.some(function(f){ return !f.c.width || !isFinite(f.ax) || !isFinite(f.ay); })) bad.push(k + ':' + S + ' frames'); if (__ED[k].top && !fr.icon) bad.push(k + ':' + S + ' icon'); } catch (e) { bad.push(k + ':' + S + ' ' + e.message); } }); [20, 40, 56, 130].forEach(function(px){ try { __EI(k, px); } catch (e) { bad.push(k + ' icon' + px + ' ' + e.message); } }); }); return bad; })(%s)" % kj)
            print('stress bad', r)
        if 'battle' in modes:
            await pg.mouse.click(180, 320); await pg.wait_for_timeout(400)
            mi = int(sys.argv[3]) if len(sys.argv) > 3 else 0
            await pg.evaluate("__td.unlockAll(); Object.keys(__td.EN).forEach(function(k){ __td.SAVE().seen[k]=1; }); __td.start(%d,'normal')" % mi)
            await pg.wait_for_timeout(2500)
            info = await pg.evaluate("""(function(list){ var g=__td.G(), d=%f; list.forEach(function(k,i){ var E=__td.EN[k]; var e=__SP(k,0,d,{hp:1e7,mhp:1e7,spd:0.02}); if(i%%6===3) e.hitT=0.09; d += (0.7 + E.size*0.75) * %f; }); return {n:g.enemies.length, S:__td.view().S}; })(%s)""" % (float(sys.argv[4]) if len(sys.argv) > 4 else 0.5, float(sys.argv[5]) if len(sys.argv) > 5 else 1.0, json.dumps([k for k in keys if k != 'submarine_sub'])))
            print(info)
            await pg.wait_for_timeout(2500)
            await pg.screenshot(path=OUT + '/battle%d.png' % mi); print('saved', OUT + '/battle%d.png' % mi)
        print('errors', len(errs)); [print(e) for e in errs[:10]]
        await b.close()
asyncio.run(main())
