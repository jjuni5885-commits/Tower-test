# XTRA 화면 캡처(다듬기용): python3 tools/xshot.py [가로x세로] [장면,...]  → shots/x_<크기>_<장면>.png
# 장면: pin, wallet, tower, stage, set, map, pause, battle, spawn, fab
import sys, os, asyncio
from playwright.async_api import async_playwright
ROOT = os.path.abspath(os.path.join(os.path.dirname(os.path.abspath(__file__)), '..'))
OUT = ROOT + '/shots/'
async def main():
    W, H = 360, 640
    if len(sys.argv) > 1 and 'x' in sys.argv[1]: W, H = map(int, sys.argv[1].split('x'))
    want = sys.argv[2].split(',') if len(sys.argv) > 2 else ['pin', 'wallet', 'tower', 'stage', 'set', 'map', 'pause', 'battle', 'spawn', 'fab']
    async with async_playwright() as p:
        b = await p.chromium.launch()
        ctx = await b.new_context(viewport={'width': W, 'height': H}, device_scale_factor=2, is_mobile=True, has_touch=True)
        pg = await ctx.new_page(); errs = []
        pg.on('pageerror', lambda e: errs.append('PAGE: ' + str(e)))
        pg.on('console', lambda m: errs.append(m.type + ': ' + m.text) if m.type == 'error' else None)
        await pg.goto('file://' + ROOT + '/out/index.html'); await pg.wait_for_timeout(1200)
        async def shot(n): await pg.screenshot(path=OUT + 'x_%dx%d_%s.png' % (W, H, n)); print('shot', n)
        if 'pin' in want: await pg.evaluate("__adm.ask()"); await pg.wait_for_timeout(400); await shot('pin'); await pg.evaluate("__td.closeModal()")
        await pg.evaluate("__adm.on(1)")
        for t in ['wallet', 'tower', 'stage', 'set']:
            if t in want: await pg.evaluate("__adm.open('%s')" % t); await pg.wait_for_timeout(500); await shot(t)
        await pg.evaluate("__td.closeModal()")
        await pg.touchscreen.tap(W / 2, H / 2); await pg.wait_for_timeout(1200)
        if 'map' in want: await shot('map')
        await pg.evaluate("__td.start(0,'normal')"); await pg.wait_for_timeout(1500)
        if 'fab' in want: await shot('fab')
        if 'pause' in want: await pg.evaluate("__meta.pause()"); await pg.wait_for_timeout(500); await shot('pause'); await pg.evaluate("__td.closeModal()")
        if 'battle' in want: await pg.evaluate("__adm.battle()"); await pg.wait_for_timeout(500); await shot('battle')
        if 'spawn' in want: await pg.evaluate("__adm.battle('spawn')"); await pg.wait_for_timeout(700); await shot('spawn')
        print('errors', errs)
        await b.close()
asyncio.run(main())
