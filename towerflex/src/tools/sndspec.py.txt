# 소리 그림: python3 tools/sndspec.py [곡 이름...]  → shots/snd_songs.png(곡 스펙트로그램과 마디별 세기), shots/snd_sfx.png(새 효과음 파형과 스펙트로그램)
# 귀로 못 듣는 환경에서 구조를 눈으로 확인하는 용도 (가락이 있는지, 빈 마디나 튀는 마디가 없는지, 폭발음이 낮은 쪽에 실렸는지)
import sys, asyncio, os, base64
import numpy as np
from scipy import signal
import matplotlib; matplotlib.use('Agg')
import matplotlib.pyplot as plt
from playwright.async_api import async_playwright
ROOT = os.path.abspath(os.path.join(os.path.dirname(os.path.abspath(__file__)), '..'))
SONGS = [a for a in sys.argv[1:] if not a.startswith('-')] or ['title', 'battle2', 'battle3', 'battle4']
if '--sfx' in sys.argv: SONGS = []
SFXS = ['flak', 'gat', 'snp', 'mslFire', 'rktSalvo', 'sinSalvo', 'blastHe', 'blastEmp', 'splash2', 'laser', 'tesla', 'navgun', 'jet', 'alarm', 'drum', 'talisman', 'sword', 'dokkaebi', 'ghost', 'transform', 'transformU', 'fuse', 'bnGold', 'bomber', 'blastS', 'tankGun', 'ball', 'flakPop', 'gatB', 'blastFire']
for a in sys.argv[1:]:
    if a.startswith('--only='): SONGS = []; SFXS = a[7:].split(',')   # 예: --only=cannon,boom (옛 소리와 견줄 때)
JS_SONG = """(async function(name){ await __snd.song(name); var u = MUSICURL(name); var ab = await (await fetch(u)).arrayBuffer(); var I = new Int16Array(ab, 44), n = I.length >> 2, o = new Int16Array(n); for (var i = 0; i < n; i++) o[i] = (I[4*i] + I[4*i+1] + I[4*i+2] + I[4*i+3]) >> 2; var b = new Uint8Array(o.buffer), s = ''; for (var j = 0; j < b.length; j += 32768) s += String.fromCharCode.apply(null, b.subarray(j, j + 32768)); return btoa(s); })"""
JS_SFX = """(function(k){ var b = __snd.buf(k); if (!b) return null; var d = b.getChannelData(0), o = new Int16Array(d.length); for (var i = 0; i < d.length; i++) o[i] = Math.max(-1, Math.min(1, d[i])) * 32767; var u = new Uint8Array(o.buffer), s = ''; for (var j = 0; j < u.length; j += 32768) s += String.fromCharCode.apply(null, u.subarray(j, j + 32768)); return btoa(s); })"""
async def main():
    os.makedirs(ROOT + '/shots', exist_ok=True)
    async with async_playwright() as p:
        b = await p.chromium.launch(args=['--autoplay-policy=no-user-gesture-required'])
        pg = await (await b.new_context(viewport={'width': 360, 'height': 640})).new_page()
        await pg.goto('file://' + ROOT + '/out/index.html'); await pg.wait_for_timeout(1500)
        for i in range(400):
            if await pg.evaluate('__snd.ready()'): break
            await pg.wait_for_timeout(150)
        info = await pg.evaluate("(function(){ var o = {}; __snd.songs().forEach(function(n){ var S = __snd.songDef(n); o[n] = { bpm: S.bpm, bars: S.bars, bpb: S.bpb || 4 }; }); return o; })()")
        if SONGS: fig, ax = plt.subplots(len(SONGS) * 2, 1, figsize=(15, 3.3 * len(SONGS)), gridspec_kw={'height_ratios': [3, 1] * len(SONGS)})
        for si, name in enumerate(SONGS):
            raw = base64.b64decode(await pg.evaluate(JS_SONG.replace('MUSICURL', '__snd.url') + "('%s')" % name))
            x = np.frombuffer(raw, dtype=np.int16).astype(np.float32) / 32768; sr = 11025; I = info[name]
            f, t, S = signal.spectrogram(x, sr, nperseg=1024, noverlap=512)
            a = ax[si * 2]; a.pcolormesh(t, f, 10 * np.log10(S + 1e-10), vmin=-75, vmax=-25, shading='auto', cmap='magma'); a.set_yscale('symlog', linthresh=200); a.set_ylim(40, 5000); a.set_title('%s  %d bpm, %d bars x %d, %.1fs' % (name, I['bpm'], I['bars'], I['bpb'], len(x) / sr), fontsize=10)
            bar = 60 / I['bpm'] * I['bpb']; n = I['bars']; rms = [float(np.sqrt(np.mean(x[int(k * bar * sr):int((k + 1) * bar * sr)] ** 2))) for k in range(n)]
            for k in range(0, n, 4): a.axvline(k * bar, color='w', lw=0.4, alpha=0.5)
            a2 = ax[si * 2 + 1]; a2.bar(np.arange(n) + 0.5, rms, width=0.9, color='#4a6fa5'); a2.set_xlim(0, n); a2.set_ylim(0, 0.2)
            print('%-8s 마디별 세기 최소 %.3f 최대 %.3f 평균 %.3f | %s' % (name, min(rms), max(rms), sum(rms) / n, ' '.join('%.2f' % r for r in rms)))
        if SONGS: plt.tight_layout(); plt.savefig(ROOT + '/shots/snd_songs.png', dpi=70); plt.close()
        await pg.evaluate("__snd.warm('all')")
        cols = 6; rows = (len(SFXS) + cols - 1) // cols
        fig, ax = plt.subplots(rows * 2, cols, figsize=(16, 2.6 * rows))
        for i, k in enumerate(SFXS):
            r64 = await pg.evaluate(JS_SFX + "('%s')" % k)
            x = np.frombuffer(base64.b64decode(r64), dtype=np.int16).astype(np.float32) / 32768; sr = 32000
            a = ax[(i // cols) * 2][i % cols]; a.plot(np.arange(len(x)) / sr, x, lw=0.4, color='#223'); a.set_ylim(-1, 1); a.set_title(k, fontsize=9); a.set_xticks([])
            f, t, S = signal.spectrogram(x, sr, nperseg=512, noverlap=384)
            a = ax[(i // cols) * 2 + 1][i % cols]; a.pcolormesh(t, f, 10 * np.log10(S + 1e-10), vmin=-80, vmax=-25, shading='auto', cmap='magma'); a.set_yscale('symlog', linthresh=300); a.set_ylim(40, 14000)
            lowE = float(S[f < 300].sum() / (S.sum() + 1e-12))
            print('%-11s %.2fs 300Hz 아래 에너지 %2d%%' % (k, len(x) / sr, round(lowE * 100)))
        plt.tight_layout(); plt.savefig(ROOT + '/shots/snd_sfx.png', dpi=62); plt.close()
        await b.close()
    print('그림:', ROOT + '/shots/snd_songs.png', ROOT + '/shots/snd_sfx.png')
asyncio.run(main())
