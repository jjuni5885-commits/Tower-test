# 기존 적 그림 픽셀 해시: python3 tools/en4_hash.py <out.json> [warm]   (warm 이면 4장 적을 먼저 만든 뒤 해시 → 상태 누수 확인)
import sys, asyncio, json, os
from playwright.async_api import async_playwright
ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
OLD = ['goblin','orc','wolf','knight','bat','stalker','shaman','spider','skeleton','troll','golemL','golemM','golemS','wyvern','wraith','wagon','warlock','iceslime','ember','mirror','bunny','guard','grumsh','arachne','yotun','barkan','yotun2','twinking']
CH4 = ['dokkaebi','wisp','egg','maiden','gumiho','saja','bulgasari','jangsan','eoduksini','tal','horangi','crows','mulgwisin','pirate','duryeok','cheonho','imugi','yong']
JS = r"""
(function(keys, warm){
  function h(c){ var d=c.getContext('2d').getImageData(0,0,c.width,c.height).data, a=2166136261>>>0; for(var i=0;i<d.length;i++){ a^=d[i]; a=Math.imul(a,16777619)>>>0; } return c.width+'x'+c.height+':'+a.toString(16); }
  var out={}, Ss=[24,36,40,57,83,130];
  if (warm) warm.forEach(function(k){ if (__E4.D[k]) { __EF(k, 36); __EF(k, 83); } });
  keys.forEach(function(k){ Ss.forEach(function(S){
    var fr=__EF(k,S), r=[fr.n, Math.round(fr.h*1000)];
    fr.fr.forEach(function(f){ r.push(h(f.c)+'@'+f.ax+','+f.ay); }); fr.wh.forEach(function(f){ r.push(h(f.c)+'@'+f.ax+','+f.ay); });
    if(fr.kneel) r.push('K'+h(fr.kneel.c)+'@'+fr.kneel.ax+','+fr.kneel.ay); if(fr.icon) r.push('I'+h(fr.icon.c)+'@'+fr.icon.ax+','+fr.icon.ay);
    out[k+':'+S]=r; }); });
  return out;
})
"""
async def main():
    dst = sys.argv[1]; warm = len(sys.argv) > 2
    async with async_playwright() as p:
        b = await p.chromium.launch()
        pg = await b.new_page(viewport={'width': 360, 'height': 640})
        errs = []
        pg.on('pageerror', lambda e: errs.append(str(e)))
        await pg.goto('file://' + ROOT + '/out/en4_test.html')
        await pg.wait_for_timeout(1200)
        r = await pg.evaluate(JS + '(%s, %s)' % (json.dumps(OLD), json.dumps(CH4 if warm else None)))
        json.dump(r, open(dst, 'w'), sort_keys=True)
        print('hashed', len(r), 'errors', errs)
        await b.close()
asyncio.run(main())
