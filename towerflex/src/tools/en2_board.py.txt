# EN2 그림판: python3 tools/en2_board.py <out_prefix> [keys(콤마)] [frames_keys(콤마)]
# big(130px 아이콘), small(40px 아이콘 + 3배 확대), line(전투 크기 S=36, 72 로 기존 적과 나란히), frames(걷기 장면 + 맞음 + 뒤집기)
import sys, asyncio, base64, json, os
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import en2_hook
from playwright.async_api import async_playwright
CH2 = ['zombie', 'ghoul', 'phantom', 'pumpkin', 'bats', 'witch', 'mummy', 'scarab', 'werewolf', 'vampire', 'coffin', 'doll', 'gargoyle', 'vampbat', 'pumpking', 'count', 'reaper']
OLD = ['goblin', 'orc', 'wolf', 'knight', 'bat', 'stalker', 'shaman', 'spider', 'skeleton', 'troll', 'golemL', 'wyvern', 'wraith', 'wagon', 'warlock', 'iceslime', 'ember', 'mirror', 'bunny', 'guard', 'grumsh', 'twinking']
JS_ICON = r"""
(function(keys, px, cols, zoom){
  var rows = Math.ceil(keys.length / cols), c = document.createElement('canvas'); c.width = cols * px * zoom; c.height = rows * px * zoom; var x = c.getContext('2d');
  x.imageSmoothingEnabled = false;
  var BG = ['#2a2444', '#3c4a46', '#8a7f9c', '#56406a'];
  return Promise.all(keys.map(function(k){ return new Promise(function(res){ var im = new Image(); im.onload = function(){ res(im); }; im.src = __EI(k, px); }); })).then(function(ims){
    ims.forEach(function(im, n){ var cx = (n % cols) * px * zoom, cy = Math.floor(n / cols) * px * zoom; x.fillStyle = BG[(n % cols + Math.floor(n / cols)) % BG.length]; x.fillRect(cx, cy, px * zoom, px * zoom); x.drawImage(im, cx, cy, px * zoom, px * zoom); });
    return c.toDataURL();
  });
})
"""
JS_LINE = r"""
(function(keys, S, perRow, zoom){
  var EF = window.__EF, gap = S * 0.2, frs = [], hmax = 0;
  keys.forEach(function(k){ if (!__td.EN[k] || !__ED[k]) return; var fr = EF(k, S), f = fr.fr[2 % fr.n]; frs.push([f, k]); hmax = Math.max(hmax, f.ay); });
  var rows = Math.ceil(frs.length / perRow), rh = hmax + S * 0.7, W = 0, cx = 10;
  var rw = []; for (var r = 0; r < rows; r++) { var w = 20; frs.slice(r * perRow, (r + 1) * perRow).forEach(function(q){ w += q[0].c.width + gap; }); rw.push(w); W = Math.max(W, w); }
  var c0 = document.createElement('canvas'); c0.width = W; c0.height = rh * rows; var x = c0.getContext('2d');
  for (r = 0; r < rows; r++) {
    x.fillStyle = r % 2 ? '#2f3b3a' : '#3a3352'; x.fillRect(0, r * rh, W, rh);
    var by = (r + 1) * rh - S * 0.35; x.fillStyle = r % 2 ? '#5a5568' : '#6f6480'; x.fillRect(0, by - S * 0.4, W, S * 0.7);
    cx = 10;
    frs.slice(r * perRow, (r + 1) * perRow).forEach(function(q){ var f = q[0]; x.fillStyle = 'rgba(10,8,20,0.35)'; x.beginPath(); x.ellipse(cx + f.ax, by, S * 0.3, S * 0.1, 0, 0, 7); x.fill(); x.drawImage(f.c, cx, by - f.ay); cx += f.c.width + gap; });
  }
  var c = document.createElement('canvas'); c.width = W * zoom; c.height = c0.height * zoom; var y = c.getContext('2d'); y.imageSmoothingEnabled = false; y.drawImage(c0, 0, 0, c.width, c.height);
  return c.toDataURL();
})
"""
JS_FR = r"""
(function(keys, S){
  var EN = __td.EN, EF = window.__EF, all = [], CW = [], RY = [0], W = 0;
  keys.forEach(function(k, r){ var fr = EF(k, EN[k].boss ? S * 0.55 : S); all.push(fr); var w = 0, h = 0; fr.fr.forEach(function(f){ w = Math.max(w, f.c.width); h = Math.max(h, f.c.height); }); CW.push(w + 6); RY.push(RY[r] + h + 8); W = Math.max(W, (w + 6) * 10); });
  var c = document.createElement('canvas'); c.width = W; c.height = RY[keys.length]; var x = c.getContext('2d');
  all.forEach(function(fr, r){
    x.fillStyle = r % 2 ? '#3c4a46' : '#3a3352'; x.fillRect(0, RY[r], W, RY[r + 1] - RY[r]);
    var hmax = 0; fr.fr.forEach(function(f){ hmax = Math.max(hmax, f.ay); });
    for (var i = 0; i < 10; i++) {
      var f = i < 8 ? fr.fr[i % fr.n] : fr.fr[1], w = fr.wh[i < 8 ? i % fr.n : 1], bx = i * CW[r] + CW[r] / 2, by = RY[r] + hmax + 4;
      x.save(); x.translate(bx, by); if (i === 9) x.scale(-1, 1);
      x.drawImage(f.c, -f.ax, -f.ay);
      if (i === 8) { x.globalAlpha = 0.9; x.drawImage(w.c, -w.ax, -w.ay); }
      x.restore();
      x.fillStyle = 'rgba(255,255,255,0.5)'; x.fillRect(bx - 1, by, 2, 2);
    }
  });
  return c.toDataURL();
})
"""
def save(d, path): open(path, 'wb').write(base64.b64decode(d.split(',')[1])); print(path)
async def main():
    pre = sys.argv[1]
    keys = sys.argv[2].split(',') if len(sys.argv) > 2 and sys.argv[2] else CH2
    fkeys = sys.argv[3].split(',') if len(sys.argv) > 3 else keys
    src = en2_hook.make()
    async with async_playwright() as p:
        b = await p.chromium.launch()
        pg = await b.new_page(viewport={'width': 360, 'height': 640})
        errs = []
        pg.on('pageerror', lambda e: errs.append(str(e)))
        pg.on('console', lambda m: errs.append(m.type + ': ' + m.text) if m.type == 'error' else None)
        await pg.goto('file://' + src)
        await pg.wait_for_timeout(1200)
        have = await pg.evaluate("Object.keys(__ED)")
        keys = [k for k in keys if k in have]; fkeys = [k for k in fkeys if k in have]
        save(await pg.evaluate(JS_ICON + '(%s, %d, %d, %d)' % (json.dumps(keys), 130, 6, 1)), pre + '_big.png')
        save(await pg.evaluate(JS_ICON + '(%s, %d, %d, %d)' % (json.dumps(keys), 40, 9, 3)), pre + '_small.png')
        save(await pg.evaluate(JS_LINE + '(%s, %d, %d, %d)' % (json.dumps(keys + OLD), 36, 13, 2)), pre + '_line.png')
        save(await pg.evaluate(JS_LINE + '(%s, %d, %d, %d)' % (json.dumps(keys), 72, 9, 1)), pre + '_line72.png')
        for i in range(0, len(fkeys), 6):
            save(await pg.evaluate(JS_FR + '(%s, %d)' % (json.dumps(fkeys[i:i + 6]), 100)), pre + '_frames%d.png' % (i // 6))
        print('errors', errs)
        await b.close()
asyncio.run(main())
