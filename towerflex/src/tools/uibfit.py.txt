# [UIB] 전투 화면 맞춤 검사: python3 tools/uibfit.py [크기들 쉼표] [--shots]
# fitcheck.py와 같은 기준(화면 안, 스크롤 없음, 카드 밖으로 안 삐져나옴, 단추 글자 안 넘침, 단추가 안 가려짐)을 전투 화면 상태에 적용:
#  위 막대와 아래 막대(규칙 칩 포함), 건설 쟁반(안 고름, 고름, 물 칸, 못 짓는 탑 고름), 탑 쟁반(1단계, 갈래 선택, 갈래 고름, 궁극, 최고 단계, 갈래 잠금, 궁극 잠금, 합체, 합체 고름, 합체 안내),
#  규칙 설명, 일시 정지(규칙 + 관리자 단추), 승리 결과(확정 보상 + 해금 안내 + 별 + 소환권). 쟁반은 고른 칸이나 탑을 가리지 않는지도 본다
# 안전 영역 흉내(노치 30px, 제스처 바 24px): 360x742s. fitcheck.py는 화면 갈래(UIA)가 고치는 중이라 따로 둠
import sys, os, asyncio, re
from playwright.async_api import async_playwright
ROOT = os.path.abspath(os.path.join(os.path.dirname(os.path.abspath(__file__)), '..'))
OUT = ROOT + '/shots/'
SIZES = ['320x568', '360x560', '360x640', '360x660', '360x742', '360x780', '375x667', '390x844', '393x852', '412x732', '412x915', '360x742s']
SHOT_SIZES = {'320x568', '360x640', '412x915'}
SRC = open(os.path.join(ROOT, 'tools', 'fitcheck.py'), encoding='utf-8').read()
CHECK = re.search(r'CHECK = r"""(.*?)"""', SRC, re.S).group(1)          # 검사 본문은 fitcheck.py의 것을 그대로 씀
FAKE = re.search(r'REAL = r"""(.*?)"""', open(os.path.join(ROOT, 'tools', 'uibflow.py'), encoding='utf-8').read(), re.S).group(1)
ALLOW = ['.tgrid.many']

async def run_size(p, spec, shots):
    safe = spec.endswith('s'); W, H = map(int, spec.rstrip('s').split('x'))
    b = await p.chromium.launch()
    ctx = await b.new_context(viewport={'width': W, 'height': H}, device_scale_factor=2, is_mobile=True, has_touch=True)
    pg = await ctx.new_page(); errs = []; res = []
    pg.on('pageerror', lambda e: errs.append('PAGE: ' + str(e)))
    pg.on('console', lambda m: errs.append(m.type + ': ' + m.text) if m.type == 'error' else None)
    if safe:
        cdp = await ctx.new_cdp_session(pg)
        await cdp.send('Emulation.setSafeAreaInsetsOverride', {'insets': {'top': 30, 'bottom': 24, 'left': 0, 'right': 0}})
    await pg.goto('file://' + ROOT + '/out/index.html'); await pg.wait_for_timeout(1300)
    async def ev(js, arg=None): return await pg.evaluate(js, arg) if arg is not None else await pg.evaluate(js)
    async def check(name, sel, wait=90, cover=None, edge=False):
        await pg.wait_for_timeout(wait)
        r = await ev(CHECK, [sel, ALLOW, edge])
        if cover is not None:   # 쟁반이 고른 칸(탑이면 머리 0.5칸 더)을 가리지 않는지
            cv = await ev("""(function(a){var v=__td.view(), cs=v.S/v.dpr, y0=(v.oy+a[1]*v.S)/v.dpr-(a[2]?cs*0.5:0), y1=(v.oy+(a[1]+1)*v.S)/v.dpr, t=document.getElementById('tray').getBoundingClientRect(); return (y1<=t.top+1||y0>=t.bottom-1) ? '' : '쟁반이 가림 '+[y0,y1,t.top,t.bottom].map(Math.round).join('/');})""", cover)
            if cv: r['fails'].append(cv)
        ok = not r['fails']; res.append((name, ok, r))
        if not ok: print('  실패 %-14s %s' % (name, r['fails'][:4]))
        if shots and spec in SHOT_SIZES: await pg.screenshot(path=OUT + 'uibfit_%s_%s.png' % (spec, re.sub(r'\W+', '_', name)))
        return r
    async def tap_sel(sel, idx=0, wait=380):
        r = await ev("""(function(a){var e=document.querySelectorAll(a[0])[a[1]]; if(!e) return null; var b=e.getBoundingClientRect(); var x=b.left+b.width/2, y=b.top+b.height/2, t=document.elementFromPoint(x,y); return {x:x,y:y,hit:!!t&&(t===e||e.contains(t))};})""", [sel, idx])
        if not r: res.append(('탭 ' + sel, False, {'fails': ['없음']})); print('  없음', sel); return None
        if not r['hit']: res.append(('탭 ' + sel, False, {'fails': ['가려짐']})); print('  가려짐', sel)
        await pg.touchscreen.tap(r['x'], r['y']); await pg.wait_for_timeout(wait)
        return r
    async def picks():
        for i in range(4):
            if await ev("__td.st().kind==='pick' && !!document.querySelector('.pkc')"): await tap_sel('.pkc', 0, 1100)
            elif await ev("__td.st().kind==='card' && !!document.getElementById('bOk')"): await tap_sel('#bOk', 0, 500)
            else: break
    await pg.add_style_tag(content='#tray, #uRuleTip { animation: none !important; }')   # 움직임 연출 중간 위치를 재지 않게
    await ev(FAKE)
    await ev("var s=__td.SAVE(); s.tut.done=1; __td.unlockAll(); __td.ownAll(); delete s.rec.c1s03; if (s.tkRew) delete s.tkRew.c1s03r; 1")
    LINE = ['archer', 'catapult', 'flak', 'frost', 'ballista', 'gatling', 'warship', 'turtle']
    await ev("__td.start(5,'normal')"); await pg.wait_for_timeout(1300); await picks()
    await ev("(function(a){var g=__td.G(); g.towersOk=a; g.gold=9000; g.maxLv=6; g.fuse=true; g.attr=g.attr||{}; g.attr.archer='fire'; g.M.ruleName='짙은 안개와 밤'; g.M.ruleDesc='안개가 끼어 모든 탑의 사거리가 15% 줄어들고, 탑 사거리 밖의 적은 보이지 않습니다.'; g.rule=g.rule||{id:'fog'}; __uib.hud(); __uib.fit();})", LINE)
    await pg.wait_for_timeout(300)
    # 판, 위 막대, 아래 막대
    L = await ev("""(function(){var v=__td.view(), g=__td.G(), s=v.S/v.dpr, hud=document.getElementById('hud').getBoundingClientRect(), bot=document.getElementById('bottom').getBoundingClientRect(), o=[];
      if ((v.oy/v.dpr) < hud.bottom-0.6 || (v.oy+v.S*g.rows)/v.dpr > bot.top+0.6) o.push('판이 막대와 겹침');
      if (v.ox < -0.5) o.push('판이 화면 밖');
      var a=[].slice.call(document.querySelectorAll('#hud .hl > *, #hud .hr > *')).filter(function(e){return getComputedStyle(e).display!=='none';}).map(function(e){return e.getBoundingClientRect();}).sort(function(p,q){return p.left-q.left;});
      for (var i=1;i<a.length;i++) if (a[i].left < a[i-1].right - 0.5) o.push('위 막대 겹침 '+i);
      if (a.length && (a[0].left < 0 || a[a.length-1].right > innerWidth + 0.5)) o.push('위 막대 화면 밖');
      return {fails:o, cell:+s.toFixed(1), side:+(v.ox/v.dpr).toFixed(1), c:document.body.classList.contains('uibC')};})()""")
    res.append(('판 배치', not L['fails'], L));
    if L['fails']: print('  실패 판 배치', L['fails'])
    await check('위 막대', '#hud', 100, None, True)
    await check('아래 막대', '#bottom', 0, None, True)
    await ev("__uib.wave()")
    bb = await ev("""(function(){var p=document.getElementById('__fitProbe'), sab=p?parseFloat(getComputedStyle(p).paddingBottom)||0:0, o=[];
      document.querySelectorAll('#bottom .sk, #bWave, #bPause, #bSpeed, #hRule').forEach(function(b){var r=b.getBoundingClientRect(), t=document.elementFromPoint(r.left+r.width/2, r.top+r.height/2);
        if (!(t&&(t===b||b.contains(t))) || r.bottom > innerHeight - Math.max(sab, 6) + 0.5 || r.top < 0 || r.width < 28 || r.height < 28) o.push([b.id, Math.round(r.top), Math.round(r.bottom), Math.round(r.width)]);}); return o;})()""")
    res.append(('막대 단추', not bb, {'fails': bb}))
    if bb: print('  실패 막대 단추', bb)
    cells = await ev("""(function(){var g=__td.G(), land=[], water=[]; for (var r=0;r<g.rows;r++) for (var c=0;c<g.cols;c++){ var i=r*g.cols+c, o=g.obsAt[i]; if (o>=0 && g.obs[o].alive) continue; if (g.kind[i]==='g') land.push([c,r]); else if (g.kind[i]==='water' && !(g.seaCell&&g.seaCell[i])) water.push([c,r]); } return {land:land, water:water};})()""")
    # 건설 쟁반: 위쪽, 가운데, 아래쪽 칸에서
    n = len(cells['land'])
    for nm, cc in [('위', cells['land'][1]), ('가운데', cells['land'][n // 2]), ('아래', cells['land'][-2])]:
        await ev("(function(a){__uib.build(a[0],a[1]);})", cc); await check('건설 ' + nm, '#tray', 60, [cc[0], cc[1], 0])
        await ev("(function(a){__uib.build(a[0],a[1],'archer');})", cc); await check('건설 고름 ' + nm, '#tray', 60, [cc[0], cc[1], 0])
    cc = cells['land'][n // 2]
    await ev("(function(a){__uib.build(a[0],a[1],'warship');})", cc); await check('건설 못 짓는 탑', '#tray', 60, [cc[0], cc[1], 0])
    wc = cells['water'][len(cells['water']) // 2]
    await ev("(function(a){__uib.build(a[0],a[1],'turtle');})", wc); await check('건설 물 칸', '#tray', 60, [wc[0], wc[1], 0])
    # 14종 편성(편성 화면이 없을 때 가진 탑 전부): 두 줄 옆으로 넘기기
    await ev("(function(a){var g=__td.G(); g.towersOk=a.concat(['arcane','bell','flame','alchemy','cannon','mint']); __uib.build(a.c,a.r);})".replace('a.c,a.r', '%d,%d' % (cc[0], cc[1])), LINE); await check('건설 14종', '#tray', 60, [cc[0], cc[1], 0])
    await ev("(function(a){__td.G().towersOk=a; __uib.close();})", LINE)
    # 탑 쟁반: 위, 가운데, 아래 탑
    spots = [cells['land'][1], cells['land'][n // 2], cells['land'][-2]]
    await ev("""(function(a){var g=__td.G(); a.forEach(function(q, i){ var t=__td.build(i===1?'archer':'catapult', q[0], q[1]); });
      var t=g.towers[1], ds=[[1,0],[-1,0],[0,1],[0,-1],[1,1],[-1,-1]]; for (var i=0;i<ds.length;i++){ var c=t.c+ds[i][0], r=t.r+ds[i][1], k=r*g.cols+c; if (c>=0&&r>=0&&c<g.cols&&r<g.rows&&g.kind[k]==='g'&&g.towerAt[k]<0&&!(g.obsAt[k]>=0&&g.obs[g.obsAt[k]].alive)) { var b=__td.build('ballista',c,r); b.lvl=5; b.br=1; break; } } __uib.re();})""", spots)
    async def tw(i, lvl, br, maxlv=6, fuse=True, pick=None):
        await ev("(function(a){var g=__td.G(), t=g.towers[a[0]]; g.maxLv=a[3]; g.fuse=a[4]; t.lvl=a[1]; t.br=a[2]; __uib.re(); if (a[5]!=null) __uib.tower2(t, a[5]); else __uib.tower(t); })", [i, lvl, br, maxlv, fuse, pick])
    for i, nm in [(0, '위'), (1, '가운데'), (2, '아래')]:
        t = await ev("(function(i){var t=__td.G().towers[i]; return [t.c,t.r,1];})", i)
        await tw(i, 1, 0); await check('탑 1단계 ' + nm, '#tray', 60, t)
        await tw(i, 3, 0); await check('탑 갈래 선택 ' + nm, '#tray', 60, t)
        await tw(i, 3, 0, 6, True, {'br': 1}); await check('탑 갈래 고름 ' + nm, '#tray', 60, t)
        await tw(i, 5, 0); await check('탑 궁극 ' + nm, '#tray', 60, t)
        await tw(i, 5, 0, 6, True, {'br': 0}); await check('탑 궁극 고름 ' + nm, '#tray', 60, t)
    t = await ev("(function(i){var t=__td.G().towers[i]; return [t.c,t.r,1];})", 1)
    await tw(1, 6, 1); await check('탑 최고 단계', '#tray', 60, t)
    await tw(1, 3, 0, 3); await check('탑 갈래 잠금', '#tray', 60, t)
    await tw(1, 5, 1, 5); await check('탑 궁극 잠금', '#tray', 60, t)
    await tw(1, 5, 0); r = await check('탑 합체', '#tray', 60, t)
    has = await ev("!!document.querySelector('#tray .uFz')"); res.append(('합체 단추 있음', has, {'fails': [] if has else ['없음']}))
    await tw(1, 5, 0, 6, True, {'fz': 0}); await check('탑 합체 고름', '#tray', 60, t)
    await tw(1, 5, 0, 6, False); await check('탑 합체 잠금', '#tray', 60, t)
    t0 = await ev("(function(i){var t=__td.G().towers[i]; return [t.c,t.r,1];})", 0)
    await tw(0, 5, 1); await check('탑 합체 안내', '#tray', 60, t0)
    await ev("__td.G().maxLv=6; __td.G().fuse=true; __uib.close()")
    # 규칙 설명
    await ev("var t=document.getElementById('uRuleTip'); if (t && !t.classList.contains('hide')) t.click(); 1"); await pg.wait_for_timeout(200)   # 시작 때 뜬 설명이 남아 있으면 닫고 칩으로 다시 엶
    await tap_sel('#hRule', 0, 350)
    await check('규칙 설명', '#uRuleTip', 100)
    await tap_sel('#hRule', 0, 250)
    # 일시 정지: 규칙 + 관리자 단추
    await ev("__td.SAVE().admin=1; 1")
    await tap_sel('#bPause', 0, 500)
    await check('일시 정지', '#modal')
    hasA = await ev("!!document.getElementById('pAdm')"); res.append(('관리자 단추', hasA, {'fails': [] if hasA else ['없음']}))
    await tap_sel('#pRes', 0, 400)
    await ev("__td.SAVE().admin=0; 1")
    # 승리 결과: 별 + 소환권 + 확정 보상 + 해금 안내
    await ev("__td.start(2,'normal')"); await pg.wait_for_timeout(1300); await picks()
    await ev("var g=__td.G(); g.over=1; g.ev.push({t:'win'}); 1"); await pg.wait_for_timeout(2700)
    await pg.wait_for_timeout(700)
    if await ev("!!document.getElementById('rwLayer')"): await check('탑 보상 획득 연출', '#rwLayer .rwCard', 50); await tap_sel('#rwOk', 0, 400)
    await check('승리 결과(보상, 해금)', '#modal')
    full = await ev("(function(){var m=document.getElementById('modal'); return !!m.querySelector('.rgain.rw') && !!m.querySelector('.uUnl') && !!m.querySelector('#rMap');})()")
    res.append(('결과 창 보상과 해금 줄', full, {'fails': [] if full else ['없음']}))
    await b.close()
    return res, errs, L

async def main():
    args = [a for a in sys.argv[1:] if not a.startswith('--')]; shots = '--shots' in sys.argv
    sizes = args[0].split(',') if args else SIZES
    os.makedirs(OUT, exist_ok=True)
    total = 0; failed = 0; allerr = []
    async with async_playwright() as p:
        for s in sizes:
            res, errs, L = await run_size(p, s, shots)
            n = len(res); f = [x[0] for x in res if not x[1]]
            total += n; failed += len(f)
            print('[%s] 칸 %.1fpx, 좌우 여백 %.1fpx, %s  %d/%d 통과%s  콘솔 오류 %d' % (s, L['cell'], L['side'], '얇은 막대' if L['c'] else '넉넉한 막대', n - len(f), n, ('  실패: ' + ', '.join(f)) if f else '', len(errs)))
            allerr += errs
    print('전체: %d/%d 통과, 콘솔 오류 %d' % (total - failed, total, len(allerr)))
    for e in allerr[:10]: print('  ', e)
asyncio.run(main())
