# EN4 성능: python3 tools/en4_perf.py   (적 한 종류 그림 만들기 시간 ms, S = 36, 54, 72(폰 2배), 97. 캐시를 피하려고 S를 조금씩 바꿔 3번 재고 5번 재서 가운데 값/가장 빠른 값. 기계가 바쁘면 가운데 값이 튄다)
import asyncio, os
from playwright.async_api import async_playwright
ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
CH4 = ['dokkaebi','wisp','egg','maiden','gumiho','saja','bulgasari','jangsan','eoduksini','tal','horangi','crows','mulgwisin','pirate','duryeok','cheonho','imugi','yong']
async def main():
    async with async_playwright() as p:
        b = await p.chromium.launch()
        pg = await b.new_page(viewport={'width': 360, 'height': 640}, device_scale_factor=2)
        await pg.goto('file://' + ROOT + '/out/en4_test.html'); await pg.wait_for_timeout(1200)
        r = await pg.evaluate("""(function(keys){ var o={}; ['goblin','wolf','barkan','twinking'].concat(keys).forEach(function(k){ var t=[]; [36, 54, 72, 97].forEach(function(s){ var m=[]; for (var j=0;j<5;j++){ var t0=performance.now(); __EF(k, s + 1 + j); m.push(performance.now()-t0); } m.sort(function(a,b){return a-b;}); t.push(+m[2].toFixed(1) + "/" + m[0].toFixed(1)); }); o[k]=t; }); return o; })(%s)""" % repr(CH4))
        worst = 0
        for k, v in r.items():
            print('%-10s' % k, v); 
            if k in CH4: worst = max(worst, max(float(q.split("/")[1]) for q in v))
        print('4장 최악(가장 빠른 값 기준, S 36~97)', worst, 'ms')
        await b.close()
asyncio.run(main())
