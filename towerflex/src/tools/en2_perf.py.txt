# EN2 성능: 적 한 종류 그림 만들기(모든 장면, 캐시 비우고) 시간 ms. python3 tools/en2_perf.py [keys(콤마)] [reps]
# 여러 작업이 같이 도는 기계에서도 믿을 수 있게: 종류를 번갈아 가며 reps 번 재고 가장 빠른 값을 씀. 기존 우두머리(twinking, grumsh)와 수레(wagon)를 기준으로 같이 잼
import asyncio, json, sys, os
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import en2_hook
from playwright.async_api import async_playwright
CH2 = ['zombie', 'ghoul', 'phantom', 'pumpkin', 'bats', 'witch', 'mummy', 'scarab', 'werewolf', 'vampire', 'coffin', 'doll', 'gargoyle', 'vampbat', 'pumpking', 'count', 'reaper', 'goblin', 'wagon', 'grumsh', 'twinking']
JS = """(function(keys, S, reps){ var best = {}; keys.forEach(function(k){ best[k] = 1e9; });
  for (var r = 0; r < reps; r++) keys.forEach(function(k){ delete __ESPR[k + ':' + S]; var t0 = performance.now(); __EF(k, S); best[k] = Math.min(best[k], performance.now() - t0); });
  keys.forEach(function(k){ best[k] = +best[k].toFixed(1); }); return best; })"""
async def main():
    keys = sys.argv[1].split(',') if len(sys.argv) > 1 and sys.argv[1] else CH2
    reps = int(sys.argv[2]) if len(sys.argv) > 2 else 12
    src = en2_hook.make('/tmp/en2w/perf.html')
    async with async_playwright() as p:
        b = await p.chromium.launch()
        pg = await b.new_page(viewport={'width': 360, 'height': 640}, device_scale_factor=2)
        await pg.goto('file://' + src); await pg.wait_for_timeout(1500)
        res = {}
        sizes = [41, 72, 97, 108]
        for S in sizes:
            res[S] = await pg.evaluate(JS + '(%s, %d, %d)' % (json.dumps(keys), S, reps))
        print('%-9s' % 'S', sizes)
        worst = 0
        for k in keys:
            row = [res[S][k] for S in sizes]
            if k in CH2[:17]: worst = max(worst, max(row[:3]))
            print('%-9s' % k, row, '' if max(row[:3]) <= 30 else '  <-- 30ms 넘음 (S 97 이하)')
        print('2장 적 가장 느린 값 (S 97 이하)', worst, 'ms')
        await b.close()
asyncio.run(main())
