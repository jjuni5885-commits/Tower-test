# 다국어 점검: 언어마다 화면 흐름을 손가락 탭(필요하면 JS 클릭)으로 돌며
#  (1) 화면에 남은 한글(DOM 글자, 캔버스 fillText/strokeText 글자, 잠깐 떴다 사라지는 알림, CSS content, 문서 제목)
#  (2) 글자 넘침(버튼, 칩, 탭, 제목 등이 제 칸이나 화면 밖으로 삐져나옴)
#  (3) 사전 검사(자리표시 {0}{s} 짝, 한글 남음, 가운데 점), 데이터 전체 점검(__i18n.audit)
# 를 모아 보여 줌. 언어 고르는 목록의 '한국어'(translate="no")는 뺌
# 사용: python3 tools/i18ncheck.py [언어들|all] [가로x세로] [흐름들|all] [--shots 언어들] [--quiet]
#   예: python3 tools/i18ncheck.py all 360x640
#       python3 tools/i18ncheck.py en,zh,th 360x640 title,map,stage,battle --shots en,zh
#       python3 tools/i18ncheck.py touchflow en 360x640     (기존 touchflow.py 를 그 언어로 한 번)
# 흐름: title map stage stages story battle tray upgrade skills wave pause settings picker win research codex lose endless boss
# 결과: 화면 요약 + shots/i18n_<언어>.json, 캡처 shots/i18n_<언어>_<이름>.png (--shots 로 고른 언어만)
# 언어는 localStorage 'lastarrow.lang' 으로 정함(게임은 자동 시험 브라우저에서 저장된 언어가 없으면 한국어)

HOOK = r"""
(function(){
  var HAN = /[ᄀ-ᇿㄱ-ㆎ가-힣]/;
  try { localStorage.setItem('lastarrow.lang', '%LANG%'); } catch (e) {}
  window.__cvKo = {}; window.__cvAll = {}; window.__koSeen = {};
  var P = CanvasRenderingContext2D.prototype, f = P.fillText, s = P.strokeText;
  function rec(t) { if (typeof t !== 'string') return; if (HAN.test(t)) __cvKo[t] = 1; if (/[^\d\s+\-!.,%x×]/.test(t)) __cvAll[t] = 1; }
  P.fillText = function (t) { rec(t); return f.apply(this, arguments); };
  P.strokeText = function (t) { rec(t); return s.apply(this, arguments); };
  function chk(n) {
    if (n.nodeType === 3) { var p = n.parentNode; if (!p || p.nodeName === 'SCRIPT' || p.nodeName === 'STYLE') return; if (p.closest && p.closest('[translate="no"]')) return; if (HAN.test(n.data)) __koSeen[n.data.trim()] = (p.id || p.className || p.nodeName); }
    else if (n.nodeType === 1 && n.nodeName !== 'SCRIPT' && n.nodeName !== 'STYLE') { var w = document.createTreeWalker(n, NodeFilter.SHOW_TEXT), t; while ((t = w.nextNode())) chk(t); }
  }
  document.addEventListener('DOMContentLoaded', function () {
    new MutationObserver(function (list) { var nodes = []; list.forEach(function (m) { if (m.type === 'characterData') nodes.push(m.target); else m.addedNodes.forEach(function (a) { nodes.push(a); }); }); setTimeout(function () { nodes.forEach(chk); }, 0); })
      .observe(document.body, { childList: true, subtree: true, characterData: true });
  });
})();
"""
SCAN = r"""
(function(){
  var HAN = /[ᄀ-ᇿㄱ-ㆎ가-힣]/, vis = [], hid = [], pse = [], W = innerWidth, H = innerHeight;
  function shown(el) { if (!el) return false; if (el.closest('.hide')) return false; var cs = getComputedStyle(el); if (cs.display === 'none' || cs.visibility === 'hidden') return false; for (var a = el; a && a.nodeType === 1; a = a.parentElement) if (+getComputedStyle(a).opacity === 0) return false; var r = el.getBoundingClientRect(); return r.width > 0 && r.height > 0; }
  function path(el) { var a = []; for (var i = 0; el && el.nodeType === 1 && i < 3; i++, el = el.parentElement) a.unshift(el.nodeName.toLowerCase() + (el.id ? '#' + el.id : '') + (typeof el.className === 'string' && el.className.trim() ? '.' + el.className.trim().split(/\s+/).slice(0, 2).join('.') : '')); return a.join(' > '); }
  var w = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT), t;
  while ((t = w.nextNode())) {
    var p = t.parentNode; if (!p || p.nodeName === 'SCRIPT' || p.nodeName === 'STYLE') continue;
    if (!HAN.test(t.data) || (p.closest && p.closest('[translate="no"]'))) continue;
    (shown(p) ? vis : hid).push([t.data.trim().slice(0, 80), path(p)]);
  }
  document.querySelectorAll('body *').forEach(function (el) {
    ['::before', '::after'].forEach(function (ps) { var c = getComputedStyle(el, ps).content; if (c && c !== 'none' && HAN.test(c) && shown(el)) pse.push([c, path(el) + ps]); });
    ['placeholder', 'title', 'aria-label'].forEach(function (a) { var v = el.getAttribute(a); if (v && HAN.test(v)) vis.push([v, path(el) + '@' + a]); });
  });
  // 넘침: 버튼, 칩, 탭, 제목 등
  var SEL = 'button, .btn, .tab, .tc, .tc .nm, .uc, .uc b, .uc span, .uc strong, .srow, .srow span, .lrow, .lopt, .ci, .ci span, .pt, .mt, .ms, .rt, .b1, .b2, .band, .node .nn, #toast, #tut, .sk em, #bWave, #bWave .wl, .ib, .stat, .xbtn, .tx, .th > b, .tt b, .tt span, .tinfo, .rgain, .newt, .ritem, .ritem b, .ritem em, .etag, .enm, .est, .who, #stSkip, .snext, .l0, .l1, .l2, .l3, .tap, .mtt, .sub, .maxed, .cbr b, .cdet > b, .rgrid span, .rgrid b, .ld, .ver, .mnum, .nb, .dinfo, .recs span, .rsub, .rs, .lhead, .lnote, .mbtns, .pbtns, .trow, .tg, #bossbar .bn';
  var ovf = [], seen = new Set();
  document.querySelectorAll(SEL).forEach(function (el) {
    if (seen.has(el) || !shown(el) || el.matches('.sk')) return; seen.add(el);   // .sk 단추는 이름표가 원래 단추보다 넓음(이름표끼리 겹침은 아래에서)
    var txt = (el.innerText || '').replace(/\s+/g, ' ').trim(); if (!txt) return;
    var cs = getComputedStyle(el), r = el.getBoundingClientRect(), why = [];
    if (el.getAnimations && el.getAnimations().some(function (a) { return a.playState === 'running' && !(a.effect && a.effect.getComputedTiming && a.effect.getComputedTiming().iterations === Infinity); })) return; // 튀어나오는 연출 중(크기가 잠깐 커짐)
    if (cs.display !== 'inline' && el.scrollWidth > el.clientWidth + 1 && el.clientWidth > 0) why.push('넘침 ' + el.scrollWidth + '>' + el.clientWidth);
    if (cs.display !== 'inline' && (cs.overflowY === 'hidden' || cs.overflow === 'hidden') && el.scrollHeight > el.clientHeight + 2 && el.clientHeight > 0) why.push('세로 잘림 ' + el.scrollHeight + '>' + el.clientHeight);
    if (r.left < -2 || r.right > W + 2) why.push('화면 밖 ' + Math.round(r.left) + '~' + Math.round(r.right));
    var pa = el.parentElement;
    if (pa && cs.position !== 'absolute' && cs.position !== 'fixed' && pa !== document.body) { var q = pa.getBoundingClientRect(), pcs = getComputedStyle(pa); if (q.width > 0 && pcs.display !== 'inline' && (r.left < q.left - 2 || r.right > q.right + 2)) why.push('부모 밖 ' + Math.round(r.left - q.left) + '/' + Math.round(q.right - r.right)); }
    if (why.length) ovf.push([path(el), txt.slice(0, 50), why.join(', ')]);
  });
  // 기술 이름표끼리 겹침
  var lab = Array.prototype.filter.call(document.querySelectorAll('.sk em'), shown).map(function (e) { return [e, e.getBoundingClientRect()]; });
  for (var i = 1; i < lab.length; i++) if (lab[i - 1][1].right > lab[i][1].left + 1) ovf.push([path(lab[i][0]), lab[i - 1][0].innerText + ' / ' + lab[i][0].innerText, '이름표 겹침 ' + Math.round(lab[i - 1][1].right - lab[i][1].left)]);
  var cv = Object.keys(window.__cvKo || {}); window.__cvKo = {};
  var seenKo = Object.keys(window.__koSeen || {}).map(function (k) { return [k.slice(0, 80), window.__koSeen[k]]; }); window.__koSeen = {};
  var title = HAN.test(document.title) ? [document.title] : [];
  return { vis: vis, hid: hid, pse: pse, ovf: ovf, cv: cv, seen: seenKo, title: title, miss: (window.__i18nMissing || []).slice() };
})()
"""

def lint_dict():
    js = r"""
const fs=require('fs'),vm=require('vm');const s=fs.readFileSync(process.argv[1],'utf8');const c={};vm.createContext(c);vm.runInContext(s+';this.R=I18N_ROWS;this.C=I18N_COLS;',c);
const HAN=/[ᄀ-ᇿㄱ-ㆎ가-힣]/,ph=x=>(x.match(/\{(\d|s|t)\}/g)||[]).sort().join('');const err=[],empty={},keys=new Set();
c.R.forEach(r=>{if(keys.has(r[0]))err.push('원문 겹침: '+r[0]);keys.add(r[0]);if(r.length!==c.C.length+1)err.push('칸 수 '+r.length+': '+r[0]);c.C.forEach((l,i)=>{const v=r[i+1];if(!v){if(r[0].indexOf('@')<0)empty[l]=(empty[l]||0)+1;return;}if(ph(v)!==ph(r[0].replace(/@.*$/,'')))err.push(l+' 자리표시: '+r[0]+' → '+v);if(HAN.test(v))err.push(l+' 한글: '+v);if(/[·・]/.test(v))err.push(l+' 가운데 점: '+v);});});
console.log(JSON.stringify({rows:c.R.length,err:err,empty:empty}));"""
    r = subprocess.run(['node', '-e', js, ROOT + '/src/i18n_data.js'], capture_output=True, text=True)

async def run_lang(b, lang, W, H, flows, shots, quiet):
    ctx = await b.new_context(viewport={'width': W, 'height': H}, device_scale_factor=2, is_mobile=True, has_touch=True)
    await ctx.add_init_script(HOOK.replace('%LANG%', lang))
    pg = await ctx.new_page()
    errs = []
    pg.on('pageerror', lambda e: errs.append('PAGE: ' + str(e)))
    pg.on('console', lambda m: errs.append(m.type + ': ' + m.text) if m.type == 'error' else None)
    await pg.goto('file://' + ROOT + '/out/index.html')
    await pg.wait_for_timeout(1300)
    R = {'lang': lang, 'vis': {}, 'hid': {}, 'cv': {}, 'seen': {}, 'pse': {}, 'title': [], 'ovf': {}, 'steps': [], 'fail': [], 'err': errs}
    async def ev(js): return await pg.evaluate(js)
    async def wait(ms): await pg.wait_for_timeout(ms)
    async def tap_xy(x, y, ms=300): await pg.touchscreen.tap(x, y); await wait(ms)
    async def tap_el(sel, ms=350, idx=0):
        r = await ev("""(function(){var e=document.querySelectorAll(%s)[%d]; if(!e) return null; var b=e.getBoundingClientRect();
          if (b.top < 0 || b.bottom > innerHeight) { e.scrollIntoView({block:'center'}); b=e.getBoundingClientRect(); }
          return {x:b.left+b.width/2, y:b.top+b.height/2};})()""" % (json.dumps(sel), idx))
        if not r: return False
        await tap_xy(r['x'], r['y'], ms); return True
    async def click(sel, ms=120, idx=0):
        ok = await ev("(function(){var e=document.querySelectorAll(%s)[%d]; if(!e) return false; e.click(); return true;})()" % (json.dumps(sel), idx))
        await wait(ms); return ok
    async def vis(id): return await ev("(function(){var e=document.getElementById('%s'); return !!e && !e.classList.contains('hide') && getComputedStyle(e).display!=='none';})()" % id)
    async def cap(name, shot=False):
        d = await ev(SCAN)
        R['steps'].append(name)
        for k in ('vis', 'hid', 'pse'):
            for t, where in d[k]: R[k].setdefault(t, '%s @%s' % (where, name))
        for t in d['cv']: R['cv'].setdefault(t, name)
        for t, where in d['seen']: R['seen'].setdefault(t, '%s @%s' % (where, name))
        for t in d['title']: R['title'].append(t)
        for p, t, why in d['ovf']: R['ovf'].setdefault(p + ' | ' + t, '%s @%s' % (why, name))
        R['miss'] = d['miss']
        if shot and lang in shots: await pg.screenshot(path=OUT + 'i18n_%s_%s.png' % (lang, name))
    async def cards():
        n = 0
        while n < 6 and await ev("!!document.getElementById('bOk') && !document.getElementById('sModal').classList.contains('hide')"):
            await cap('card', n == 0); await tap_el('#bOk', 350); n += 1
    async def free_cells():
        return await ev("""(function(){var g=__td.G(), out=[]; for (var r=1;r<13;r++) for (var c=0;c<8;c++){ var i=r*8+c;
           if (g.kind[i]!=='g' || g.obsAt[i]>=0) continue; if (g.towers.some(function(t){return t.c===c&&t.r===r;})) continue; out.push([c,r]); } return out;})()""")
    async def cell_xy(c, r):
        return await ev("(function(){var v=__td.view(); return [(v.ox + (%d + 0.5) * v.S) / v.dpr, (v.oy + (%d + 0.5) * v.S) / v.dpr];})()" % (c, r))
    async def step(name, fn):
        if name not in flows: return
        try: await fn()
        except Exception as e: R['fail'].append('%s: %s' % (name, str(e)[:160]))
    st = {}
    # ---- 흐름 ----
    async def f_title(): await cap('title', True)
    async def f_map():
        await ev("__td.unlockAll()")
        await tap_xy(W / 2, H / 2, 1100)
        await cap('map', True)
    async def f_stage():
        for i in (0, 1, 2):
            if not await vis('sMap'): await ev("__td.closeModal()")
            await tap_el('.node', 700, i)
            await cap('stage%d' % (i + 1), i in (0, 2))
            for dk in ('easy', 'hard', 'endless', 'normal'):
                await click('.tab[data-dk="%s"]' % dk, 150); await cap('stage%d_%s' % (i + 1, dk))
            await tap_el('#bClose', 400)
    async def f_stages():
        for i in range(3, 12):
            await click('.node', 250, i); await cap('stage%d' % (i + 1)); await click('#bClose', 150)
    async def f_story():
        await tap_el('.node', 700, 0); await tap_el('#bGo', 900)
        n = 0
        while await vis('story') and n < 12:
            await cap('story%d' % n, n == 1); await tap_xy(W / 2, H * 0.78, 350); n += 1
        await wait(800)
    async def f_battle():
        if not await ev("!!__td.G()"):
            await ev("__td.start(0,'normal')"); await wait(800)

        await cap('battle', True)
    async def f_tray():
        cells = await free_cells(); c, r = cells[len(cells) // 2]; st['cell'] = (c, r)
        x, y = await cell_xy(c, r); await tap_xy(x, y, 450)
        await cap('tray', True)
        ks = await ev("Array.prototype.map.call(document.querySelectorAll('#tray .tc'), function(b){return b.getAttribute('data-k');})")
        for k in ks:
            await click('#tray .tc[data-k="%s"]' % k, 120); await cap('tray_' + k, k == 'bell')

def touchflow(lang, size):
    src = open(os.path.join(ROOT, 'tools', 'touchflow.py'), encoding='utf-8').read()
    inj = "        await ctx.add_init_script(\"try{localStorage.setItem('lastarrow.lang','%s')}catch(e){}\")\n        pg = await ctx.new_page()" % lang
    src = src.replace('        pg = await ctx.new_page()', inj, 1)
    sys.argv = [os.path.join(ROOT, 'tools', 'touchflow.py'), size]
    exec(compile(src, 'touchflow.py(' + lang + ')', 'exec'), {'__name__': '__main__', '__file__': os.path.join(ROOT, 'tools', 'touchflow.py')})
