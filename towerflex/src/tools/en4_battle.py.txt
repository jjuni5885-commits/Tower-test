# EN4 전투 캡처: python3 tools/en4_battle.py <맵 번호> <out.png> [keys(콤마)]   (시험 사본 out/en4_test.html, 소스는 안 건드림)
import asyncio, sys, os, json
from playwright.async_api import async_playwright
ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
MI = int(sys.argv[1]) if len(sys.argv) > 1 else 0
OUT = sys.argv[2] if len(sys.argv) > 2 else ROOT + '/shots/en4_battle.png'
KEYS = sys.argv[3].split(',') if len(sys.argv) > 3 else ['dokkaebi','egg','gumiho','saja','wisp','bulgasari','jangsan','eoduksini','tal','maiden','horangi','crows','mulgwisin','pirate','dokkaebi','goblin','knight']
async def main():
    async with async_playwright() as p:
        b = await p.chromium.launch()
        pg = await b.new_page(viewport={'width': 360, 'height': 640}, device_scale_factor=2)
        errs = []
        pg.on('pageerror', lambda e: errs.append('PAGE ' + str(e)))
        pg.on('console', lambda m: errs.append(m.type + ': ' + m.text) if m.type in ('error', 'warning') else None)
        await pg.goto('file://' + ROOT + '/out/en4_test.html'); await pg.wait_for_timeout(1500)
        await pg.mouse.click(180, 320); await pg.wait_for_timeout(400)
        await pg.evaluate("__td.unlockAll(); Object.keys(__td.EN).forEach(function(k){ __td.SAVE().seen[k]=1; }); __td.start(%d,'normal')" % MI)
        await pg.wait_for_timeout(2500)
        info = await pg.evaluate("""(function(list){
          var g=__td.G(), d=0.5, n=0; g.enemies.length = 0;
          list.forEach(function(k,i){ var E=__td.EN[k]; if(!E) return; var e=__SP(k,0,d,{hp:1e7,mhp:1e7,spd:0.02}); if(i%%6===3) e.hitT=0.09; d += E.boss ? 2.6 : 0.6 + E.size*0.75; n++; });
          return {n:n, S:__td.view().S};
        })(%s)""" % json.dumps(KEYS))
        print(info)
        await pg.wait_for_timeout(2500)
        await pg.screenshot(path=OUT)
        st = await pg.evaluate("__td.st().err")
        print('loop err', st, 'errors', len(errs)); [print(e) for e in errs[:10]]
        await b.close()
asyncio.run(main())
